基于审计发现的 20 个关键缺陷，我们将原方案从"策略互斥"重构为"Pipeline 组合"架构，并修复所有 P0/P1 问题。

- ✅ 架构从互斥切换到 Pipeline（可组合 WRR + Bandit）
- ✅ WRR 增加防溢出、对数归一化、并发安全
- ✅ 水位监控从全量轮询改为增量查询
- ✅ 优先级队列从纯优先级改为 WFQ 加权公平
- ✅ 新增三级降级链（WRR → Bandit → Random）

**工作量调整**：20d → 31d（+55%）

```go
func TestWRR_NoOverflow(t *testing.T) {

func TestWRR_ConcurrentUpdate(t *testing.T) {
    wrr := NewSmoothWRR(candidates)

    var wg sync.WaitGroup
    // 100 并发更新
    for i := 0; i < 100; i++ {
        wg.Add(1)
        go func() {
            defer wg.Done()
            wrr.UpdateWeights(map[int]int{1: rand.Intn(100)})
        }()
    }

    // 1000 并发选择
    for i := 0; i < 1000; i++ {
        wg.Add(1)
        go func() {
            defer wg.Done()
            wrr.Select()
        }()
    }

    wg.Wait()  // 不 panic, 不死锁
}
```

### Day 5: 优先级队列 WFQ

**任务清单**：

```
□ AUDIT-013: WFQ 加权公平实现
  - 复合评分: priority*0.7 + normalizedWait*0.3
  - 等待时间归一化（60s 饱和）
  - Deadline 自动清理

□ 单元测试
  - 高优先级不会饿死低优先级
  - 低优先级等待 60s 后一定出队
```

**验收标准**：
```go
func TestWFQ_NoStarvation(t *testing.T) {
    wfq := NewWFQ()

    // 插入 1 个低优先级
    lowEntry := wfq.Enqueue(credID=1, priority=1, timeout=5*time.Second)

    // 插入 100 个高优先级
    for i := 0; i < 100; i++ {
        wfq.Enqueue(credID=2, priority=10, timeout=5*time.Second)
    }

    // 持续出队
    start := time.Now()
    for {
        entry := wfq.Dequeue()
        if entry == lowEntry {
            elapsed := time.Since(start)
            // 低优先级在 60s 内必出队（不会饿死）
            assert.Less(t, elapsed, 60*time.Second)
            break
        }
    }
}
```

## Rollback 预案

### 场景 1: WRR 选路异常

```bash
# 秒切回 Bandit
UPDATE provider_policies
SET routing_pipeline = '{"name":"bandit","stages":[...]}'::jsonb
WHERE provider_id = <affected_provider>;

# 验证
SELECT * FROM provider_policies WHERE provider_id = <id>;
```

### 场景 2: 水位告警误报

```bash
# 临时禁用水位监控
export WATERMARK_MONITOR_ENABLED=false
systemctl restart llm-gateway-go

# 或调整阈值
UPDATE provider_policies
SET watermark_alert_warn = 0.80,  -- 从 0.70 提高
    watermark_alert_critical = 0.90;  -- 从 0.85 提高
```

### 场景 3: 排队队列死锁

```bash
# 禁用排队
export QUEUE_ENABLED=false
systemctl restart llm-gateway-go
```
