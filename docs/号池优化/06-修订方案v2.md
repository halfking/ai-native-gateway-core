## 1. 核心架构重构：Pipeline 模式

### 1.1 从互斥到组合

```go
// v1: 策略互斥（违反 AUDIT-016）
type Scheduler struct {
    strategy Strategy  // WRR 或 Bandit，二选一 ❌
}

// v2: Pipeline 组合 ✅
type Scheduler struct {
    pipeline *SelectionPipeline
}

type SelectionPipeline struct {
    stages []SelectionStage
}

type SelectionStage interface {
    // 每个 stage 都是一个过滤器/排序器/采样器
    Process(ctx context.Context, candidates []Candidate) ([]Candidate, error)
}
```

### 1.3 降级链（AUDIT-017 修复）

```go
type FallbackChain struct {
    primary   *SelectionPipeline
    fallbacks []*SelectionPipeline
}

func (fc *FallbackChain) Select(ctx context.Context, candidates []Candidate) (*Candidate, error) {
    // 1. 主策略
    if cand, err := fc.primary.Select(ctx, candidates); err == nil && cand != nil {
        return cand, nil
    }

    // 2. 降级链
    for i, fallback := range fc.fallbacks {
        slog.Warn("primary pipeline failed, trying fallback",
            "fallback_index", i,
            "fallback_name", fallback.Name(),
        )
        if cand, err := fallback.Select(ctx, candidates); err == nil && cand != nil {
            return cand, nil
        }
    }

    // 3. 最后兜底：全局池 uniform random
    return &candidates[rand.Intn(len(candidates))], nil
}
```

## 2. WRR 实现修复

### 2.1 防溢出（AUDIT-001）

```go
type SmoothWRR struct {
    candidates   []*wrrItem
    totalWeight  int64
    gcd          int64  // 权重的最大公约数
    cycleCount   int64  // 选择周期计数
    mu           sync.RWMutex
}

func (w *SmoothWRR) Select() *Candidate {
    w.mu.Lock()
    defer w.mu.Unlock()

    if len(w.candidates) == 0 {
        return nil
    }

    // 1. 累加权重
    var best *wrrItem
    for _, item := range w.candidates {
        item.currentWeight += item.weight
        if best == nil || item.currentWeight > best.currentWeight {
            best = item
        }
    }

    // 2. 扣减总权重
    best.currentWeight -= w.totalWeight

    // 3. 周期性归零（AUDIT-001 修复）
    w.cycleCount++
    if w.cycleCount%(w.gcd*10) == 0 {
        w.resetCurrentWeights()
    }

    return &best.candidate
}

func (w *SmoothWRR) resetCurrentWeights() {
    for _, item := range w.candidates {
        item.currentWeight = 0
    }
}

func gcd(a, b int64) int64 {
    for b != 0 {
        a, b = b, a%b
    }
    return a
}
```

### 2.3 并发安全的权重更新（AUDIT-003）

```go
func (w *SmoothWRR) UpdateWeights(adjustments map[int]int) {
    w.mu.Lock()
    defer w.mu.Unlock()

    // 1. 更新 weight
    for _, item := range w.candidates {
        if newW, ok := adjustments[item.candidate.CredentialID]; ok {
            item.weight = int64(newW)
        }
    }

    // 2. 重新计算 totalWeight 和 gcd
    w.recalcTotalWeightAndGCD()
}

func (w *SmoothWRR) recalcTotalWeightAndGCD() {
    var total int64
    var g int64

    for i, item := range w.candidates {
        total += item.weight
        if i == 0 {
            g = item.weight
        } else {
            g = gcd(g, item.weight)
        }
    }

    w.totalWeight = total
    w.gcd = g
}
```

### 2.4 候选集变更同步（AUDIT-005）

```go
func (w *SmoothWRR) Reload(newCandidates []Candidate) {
    w.mu.Lock()
    defer w.mu.Unlock()

    // 保留旧状态
    oldState := make(map[int]int64)  // credentialID → currentWeight
    for _, item := range w.candidates {
        oldState[item.candidate.CredentialID] = item.currentWeight
    }

    // 重建
    w.candidates = make([]*wrrItem, len(newCandidates))
    for i, c := range newCandidates {
        w.candidates[i] = &wrrItem{
            candidate:     c,
            weight:        int64(c.Weight),
            currentWeight: oldState[c.CredentialID],  // 继承
        }
    }

    w.recalcTotalWeightAndGCD()
}
```

## 3. 权重计算修复

### 3.1 对数归一化（AUDIT-007）

```go
type WeightCalculator struct {
    rpmFactor    float64  // 0.6
    tpmFactor    float64  // 0.3
    successFactor float64  // 0.1
}

func (wc *WeightCalculator) Calculate(c *Candidate) int {
    rpm := float64(derefInt(c.RPMLimit))
    tpm := float64(derefInt(c.TPMLimit)) / 1000
    successRate := c.SuccessRate

    // 基础分数
    baseScore := rpm*wc.rpmFactor + tpm*wc.tpmFactor + successRate*100*wc.successFactor

    return max(1, int(baseScore))
}

// 对数归一化（AUDIT-007 修复）
func (wc *WeightCalculator) NormalizeLog(weights []int) []int {
    if len(weights) == 0 {
        return weights
    }

    // 1. 转对数空间
    logWeights := make([]float64, len(weights))
    for i, w := range weights {
        logWeights[i] = math.Log1p(float64(w))  // log(1+w)，避免 log(0)
    }

    // 2. 归一化到 1-100
    maxLog := slices.Max(logWeights)
    if maxLog <= 0 {
        maxLog = 1  // 防止除零
    }

    normalized := make([]int, len(weights))
    for i, lw := range logWeights {
        normalized[i] = max(1, int(lw/maxLog*100))
    }

    return normalized
}
```

type WatermarkMonitor struct {
    db               *pgxpool.Pool
    lastCheckTime    time.Time
    cachedStats      *PoolStats     // 缓存上次全量结果
    useIncrementalQuery bool
}

func (m *WatermarkMonitor) Check(ctx context.Context) (*PoolWatermark, error) {
    now := time.Now()

    var stats *PoolStats
    if m.useIncrementalQuery && now.Sub(m.lastCheckTime) < 5*time.Minute {
        // 增量模式：只查最近变更的
        delta, err := m.loadIncrementalStats(ctx, m.lastCheckTime)
        if err == nil {
            stats = m.mergeStats(m.cachedStats, delta)
        } else {
            // 降级到全量
            stats, _ = m.loadFullStats(ctx)
        }
    } else {
        // 全量模式
        stats, _ = m.loadFullStats(ctx)
        m.cachedStats = stats

    m.lastCheckTime = now
    return m.calculateWatermark(stats), nil

## 5. 并发排队缓冲优化

### 5.1 WFQ 加权公平队列（AUDIT-013）

```go
type WeightedFairQueue struct {
    heap     *priorityHeap
    mu       sync.Mutex
    notEmpty chan struct{}
}

type queueEntry struct {
    credentialID int
    priority     int
    enqueueAt    time.Time
    deadline     time.Time
    ch           chan struct{}
    index        int  // heap index
}

// 复合评分：priority * 0.7 + waitTime * 0.3
func (pq *priorityHeap) Less(i, j int) bool {
    ei, ej := (*pq)[i], (*pq)[j]

    waitI := time.Since(ei.enqueueAt).Seconds()
    waitJ := time.Since(ej.enqueueAt).Seconds()

    // 归一化等待时间（避免等待时间主导）
    normalizedWaitI := math.Min(waitI/60, 1.0)  // 60s 内线性增长，之后饱和
    normalizedWaitJ := math.Min(waitJ/60, 1.0)

    scoreI := float64(ei.priority)*0.7 + normalizedWaitI*0.3
    scoreJ := float64(ej.priority)*0.7 + normalizedWaitJ*0.3

    return scoreI > scoreJ
}

// Deadline 自动清理
func (wfq *WeightedFairQueue) cleanupExpired() {
    wfq.mu.Lock()
    defer wfq.mu.Unlock()

    now := time.Now()
    for wfq.heap.Len() > 0 {
        top := (*wfq.heap)[0]
        if top.deadline.After(now) {
            break  // 堆顶未过期，后续也不会过期
        }
        heap.Pop(wfq.heap)
        close(top.ch)  // 通知调用方超时
    }
}
```

func (s *Scheduler) Select(ctx context.Context, candidates []Candidate) (*Candidate, error) {
    ctx, span := otel.Tracer("scheduler").Start(ctx, "scheduler.Select")
    defer span.End()

    span.SetAttributes(
        attribute.String("strategy", s.pipeline.Name()),
        attribute.Int("candidate_count", len(candidates)),

    cand, err := s.pipeline.Select(ctx, candidates)

    if cand != nil {
        span.SetAttributes(
            attribute.Int("selected_credential_id", cand.CredentialID),
            attribute.Int("selected_weight", cand.Weight),
            attribute.String("selected_provider", cand.Provider),
        )
    }

    if err != nil {
        span.RecordError(err)
        span.SetStatus(codes.Error, err.Error())
    }

    return cand, err
}
