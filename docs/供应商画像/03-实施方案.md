# 供应商质量画像 - 实施方案

## 1. 概述

本文档定义供应商质量画像系统的分阶段实施方案，确保平滑上线、风险可控。

### 1.1 实施目标

- **短期**（1-2周）：基础指标采集与存储
- **中期**（3-4周）：实时画像计算与告警
- **长期**（5-8周）：与路由系统集成，智能决策

### 1.2 实施原则

1. **最小侵入**：复用现有日志，减少代码改动
2. **渐进式上线**：先观测，后决策，最后自动化
3. **可回滚**：每个阶段可独立禁用
4. **零故障**：不影响主链路性能

```go
// internal/quality/profile_calculator.go

type ProfileCalculator struct {
    db     *sqlx.DB
    redis  *redis.Client
    config *Config
}

func (pc *ProfileCalculator) CalculateProfile(providerID int64) (*QualityProfile, error) {
    // 1. 从 Redis 和 DB 获取各时间窗口指标
    metrics5m := pc.getMetrics(providerID, "5m")
    metrics1h := pc.getMetrics(providerID, "1h")
    metrics24h := pc.getMetrics(providerID, "24h")
    
    // 2. 计算四层指标
    l1 := pc.calculateAvailability(metrics5m, metrics1h, metrics24h)
    l2 := pc.calculatePerformance(metrics5m, metrics1h, metrics24h)
    l3 := pc.calculateStability(metrics24h)
    l4 := pc.calculateCostEfficiency(providerID, metrics24h)
    
    // 3. 计算综合评分
    score := pc.calculateQualityScore(l1, l2, l3, l4)
    
    // 4. 判定等级
    grade := pc.determineGrade(score)
    
    return &QualityProfile{
        ProviderID:        providerID,
        AvailabilityScore: l1.Score,
        PerformanceScore:  l2.Score,
        StabilityScore:    l3.Score,
        CostEfficiency:    l4.Score,
        QualityScore:      score,
        QualityGrade:      grade,
        // ... 其他字段
    }, nil
}
```

#### 2.3 评分算法实现

```go
func (pc *ProfileCalculator) calculateQualityScore(
    avail, perf, stab, cost float64,
) float64 {
    w := pc.config.Weights
    return avail*w.Availability + 
           perf*w.Performance + 
           stab*w.Stability + 
           cost*w.CostEfficiency
}

func (pc *ProfileCalculator) determineGrade(score float64) string {
    switch {
    case score >= 90:
        return "S"
    case score >= 80:
        return "A"
    case score >= 70:
        return "B"
    case score >= 60:
        return "C"
    default:
        return "D"
    }
}
```

```go
// internal/router/decision_engine.go (改造)

type DecisionEngine struct {
    // 现有字段...
    qualityStore *quality.ProfileStore  // 新增
}

func (de *DecisionEngine) SelectProvider(ctx context.Context, req *Request) (*Provider, error) {
    candidates := de.getCandidates(req.Model)
    
    // 过滤掉质量不达标的供应商
    qualified := de.filterByQuality(candidates)
    
    // 按质量评分调整权重
    weighted := de.adjustWeightsByQuality(qualified)
    
    // 加权随机选择
    return de.weightedRandom(weighted), nil
}

func (de *DecisionEngine) filterByQuality(candidates []*Provider) []*Provider {
    var qualified []*Provider
    
    for _, p := range candidates {
        profile := de.qualityStore.Get(p.ID)
        
        // 硬性过滤条件
        if profile.QualityScore < 60 {
            log.Warn("provider %d filtered out: score too low", p.ID)
            continue
        }
        
        if profile.ConsecutiveFailures >= 5 {
            log.Warn("provider %d filtered out: circuit open", p.ID)
            continue
        }
        
        qualified = append(qualified, p)
    }
    
    return qualified
}

func (de *DecisionEngine) adjustWeightsByQuality(providers []*Provider) []*WeightedProvider {
    var weighted []*WeightedProvider
    
    for _, p := range providers {
        profile := de.qualityStore.Get(p.ID)
        
        // 基础权重
        baseWeight := p.Priority
        
        // 根据质量评分调整
        qualityMultiplier := profile.QualityScore / 100.0
        
        // 额外奖励 S/A 级供应商
        if profile.QualityGrade == "S" {
            qualityMultiplier *= 1.2
        } else if profile.QualityGrade == "A" {
            qualityMultiplier *= 1.1
        }
        
        adjustedWeight := baseWeight * qualityMultiplier
        
        weighted = append(weighted, &WeightedProvider{
            Provider: p,
            Weight:   adjustedWeight,
        })
    }
    
    return weighted
}
```

```go
type DegradationStrategy struct {
    qualityStore *quality.ProfileStore
}

func (ds *DegradationStrategy) Apply(providers []*Provider) []*Provider {
    for _, p := range providers {
        profile := ds.qualityStore.Get(p.ID)
        
        // 降级策略
        if profile.QualityScore < 70 {
            // 降低权重到原来的50%
            p.Priority *= 0.5

            // 完全移除
            p.Enabled = false
        }
    }
    
    return providers
}
```

## 3. 灰度上线策略

### 3.1 观察模式（第1周）

- **启用**：数据采集 + 指标计算
- **禁用**：路由决策集成
- **目标**：观察指标准确性，调整阈值

### 3.2 影子模式（第2周）

- **启用**：路由决策 (shadow)
- **行为**：记录决策结果，但不实际应用
- **目标**：对比质量驱动路由 vs 现有路由的差异

### 3.3 小流量验证（第3周）

- **启用**：路由决策（10% 流量）
- **监控**：成功率、延迟、错误率变化
- **目标**：验证质量驱动路由的效果

### 3.4 全量上线（第4周）

- **启用**：所有功能 100% 流量
