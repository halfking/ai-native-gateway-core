## 📚 API 使用示例

### 示例 1: 查询供应商质量画像

```typescript
import { qualityApi, isSuccessResponse } from '@/api/quality-api-client';

async function fetchProviderQuality(providerId: number) {
  try {
    const response = await qualityApi.getProviderQuality({
      provider_id: providerId,
    });

    if (isSuccessResponse(response)) {
      console.log('供应商名称:', response.data.provider_name);
      console.log('模型数量:', response.data.models.length);
      
      response.data.models.forEach(model => {
        console.log(`${model.model_name}: ${model.quality_score} (${model.quality_grade})`);
      });
    }
  } catch (error) {
    console.error('查询失败:', error);
  }
}
```

### 示例 2: 查询质量排行榜

```typescript
async function fetchRanking() {
  try {
    const response = await qualityApi.getRanking({
      limit: 10,
      min_score: 80,
      order_by: 'quality_score',
    });

    if (isSuccessResponse(response)) {
      console.log('总数:', response.data.total);
      
      response.data.ranking.forEach(item => {
        console.log(`#${item.rank} ${item.provider_name} - ${item.model_name}: ${item.quality_score}`);
      });
    }
  } catch (error) {
    console.error('查询失败:', error);
  }
}
```

### 示例 3: 按模型过滤

```typescript
async function fetchByModel(providerId: number, modelName: string) {
  const response = await qualityApi.getProviderQuality({
    provider_id: providerId,
    model_name: modelName,
  });

  return response.data.models[0]; // 返回单个模型数据
}
```

### 示例 4: 错误处理

```typescript
import { isErrorResponse, getErrorMessage } from '@/api/quality-api-client';

async function handleQualityQuery() {
  try {
    const response = await qualityApi.getProviderQuality({
      provider_id: 999, // 不存在的供应商
    });
  } catch (error) {
    if (isErrorResponse(error)) {
      const message = getErrorMessage(error.code);
      console.error(`错误 ${error.code}: ${message}`);
      
      // 根据错误码处理
      switch (error.code) {
        case 40401:
          alert('供应商不存在');
          break;
        case 40402:
          alert('暂无质量数据');
          break;
        default:
          alert('查询失败');
      }
    }
  }
}
```

## 🎨 Vue 3 组件示例

### 供应商质量画像组件

```vue
<template>
  <div class="provider-quality">
    <h2>{{ providerName }}</h2>
    
    <div v-if="loading">加载中...</div>
    
    <div v-else-if="error" class="error">
      错误: {{ error.message }}
    </div>
    
    <div v-else-if="models.length > 0">
      <div v-for="model in models" :key="model.model_name" class="model-card">
        <h3>{{ model.model_name }}</h3>
        <div class="score">
          质量分数: {{ formatQualityScore(model.quality_score) }}
          <span :class="`grade-${model.quality_grade}`">
            {{ model.quality_grade }}级
          </span>
        </div>
        
        <div class="scores-detail">
          <div>可用性: {{ model.scores.availability }}</div>
          <div>性能: {{ model.scores.performance }}</div>
          <div>稳定性: {{ model.scores.stability }}</div>
          <div>成本效益: {{ model.scores.cost_efficiency }}</div>
        </div>
        
        <div class="timestamp">
          更新时间: {{ formatCalculatedAt(model.calculated_at) }}
        </div>
      </div>
    </div>
    
    <div v-else>暂无数据</div>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted } from 'vue';
import { qualityApi } from '@/api/quality-api-client';
import { formatQualityScore, formatCalculatedAt } from '@/types/quality-api';
import type { ModelQualityProfile } from '@/types/quality-api';

const props = defineProps<{
  providerId: number;
}>();

const loading = ref(false);
const error = ref<{ message: string } | null>(null);
const providerName = ref('');
const models = ref<ModelQualityProfile[]>([]);

onMounted(async () => {
  loading.value = true;
  error.value = null;
  
  try {
    const response = await qualityApi.getProviderQuality({
      provider_id: props.providerId,
    });
    
    providerName.value = response.data.provider_name;
    models.value = response.data.models;
  } catch (e: any) {
    error.value = { message: e.message || '查询失败' };
  } finally {
    loading.value = false;
  }
});
</script>
```

```vue
<template>

    <div class="filters">

          <th>排名</th>

          <th>模型</th>
          <th>质量分数</th>

          <th>可用性</th>
          <th>性能</th>
        </tr>
      </thead>
      <tbody>
        <tr v-for="item in ranking" :key="`${item.provider_id}-${item.model_name}`">
          <td>{{ item.rank }}</td>
          <td>{{ item.provider_name }}</td>
          <td>{{ item.model_name }}</td>
          <td>{{ formatQualityScore(item.quality_score) }}</td>
          <td :class="`grade-${item.quality_grade}`">{{ item.quality_grade }}</td>

<script setup lang="ts">
import { ref, reactive, onMounted } from 'vue';
import { qualityApi } from '@/api/quality-api-client';
import { formatQualityScore } from '@/types/quality-api';
import type { RankingItem, GetRankingParams } from '@/types/quality-api';

const loading = ref(false);
const ranking = ref<RankingItem[]>([]);

const filters = reactive<GetRankingParams>({
  limit: 20,
  order_by: 'quality_score',

});

  } catch (e) {
    console.error('查询失败:', e);
  } finally {
    loading.value = false;
  }
}

onMounted(() => {
  fetchRanking();
});
</script>
```

## 📊 类型定义说明

### 核心类型

| 类型 | 说明 | 用途 |
|------|------|------|
| `QualityScores` | 五维评分 | 展示详细评分 |
| `QualityGrade` | 质量等级 (S/A/B/C/D) | 等级标签 |
| `ModelQualityProfile` | 模型质量画像 | 单个模型数据 |
| `ProviderQualityData` | 供应商质量数据 | 供应商所有模型 |
| `RankingItem` | 排行榜项 | 排行榜单条数据 |

### 工具函数

| 函数 | 说明 | 示例 |
|------|------|------|
| `isSuccessResponse` | 判断成功响应 | `if (isSuccessResponse(res))` |
| `isErrorResponse` | 判断错误响应 | `if (isErrorResponse(err))` |
| `getErrorMessage` | 获取错误消息 | `getErrorMessage(40401)` |
| `formatQualityScore` | 格式化分数 | `formatQualityScore(95.567)` → `"95.6"` |
| `formatCalculatedAt` | 格式化时间 | ISO 8601 → `"2026/07/19 12:54"` |
