## 4. 状态管理

```ts
const hours = ref<'1' | '24' | '168'>('24')
const loading = ref(false)
const error = ref<string>('')
const data = ref<VendorCredentialErrorDetail | null>(null)

async function loadData() {
  if (!props.credentialId) {
    data.value = null
    return
  }
  loading.value = true
  error.value = ''
  try {
    data.value = await getVendorCredentialErrorDetail(props.credentialId, hours.value)
  } catch (e) {
    error.value = e instanceof Error ? e.message : '加载错误明细失败'
    data.value = null
  } finally {
    loading.value = false
  }
}

watch(() => props.credentialId, loadData, { immediate: true })
watch(hours, loadData)
```
