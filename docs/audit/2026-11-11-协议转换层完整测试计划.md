    }

func testRoundtrip(t *testing.T, srcProto, dstProto string) {
    // 1. 构建源协议请求
    srcReq := buildFixture(srcProto)
    
    // 2. 解析为IR
    ir1, err := parseByProtocol(srcReq, srcProto)
    require.NoError(t, err)
    
    // 3. 序列化为目标协议
    dstData, err := serializeByProtocol(ir1, dstProto)
    require.NoError(t, err)
    
    // 4. 解析目标协议为IR
    ir2, err := parseByProtocol(dstData, dstProto)
    require.NoError(t, err)
    
    // 5. 再序列化为目标协议
    dstData2, err := serializeByProtocol(ir2, dstProto)
    require.NoError(t, err)
    
    // 6. 验证二次往返稳定性
    assert.JSONEq(t, string(dstData), string(dstData2), 
        "Second roundtrip should be identical")
    
    // 7. 验证语义等价性
    assertSemanticEquivalent(t, ir1, ir2)
    
    // 8. 验证扩展字段保留
    assertExtensionsPreserved(t, ir1, ir2)
}

func assertSemanticEquivalent(t *testing.T, ir1, ir2 *ir.InternalRequest) {
    // 核心字段必须相同
    assert.Equal(t, ir1.Model, ir2.Model)
    assert.Equal(t, len(ir1.Messages), len(ir2.Messages))
    
    // 采样参数必须等价 (考虑协议差异)
    assertSamplingEquivalent(t, ir1, ir2)
    
    // 工具定义必须等价
    assertToolsEquivalent(t, ir1.Tools, ir2.Tools)
}

        val2, exists := ir2.Extensions[key]

        })

package ir_test

import (
    "sync"
    "testing"
    "github.com/kaixuan/llm-gateway-go/internal/ir"

// TestConcurrentConversion 测试并发转换安全性
func TestConcurrentConversion(t *testing.T) {
    const goroutines = 100
    const iterations = 1000
    
    data := []byte(openaiFixture())
    
    var wg sync.WaitGroup
    errors := make(chan error, goroutines*iterations)
    
    for g := 0; g < goroutines; g++ {
        wg.Add(1)
        go func() {
            defer wg.Done()
            for i := 0; i < iterations; i++ {
                // 并发解析
                req, err := ir.ParseOpenAI(data)
                if err != nil {
                    errors <- err
                    continue
                }
                
                // 并发序列化
                _, err = ir.SerializeAnthropic(req)
                if err != nil {
                    errors <- err
                }
            }
        }()
    }
    
    wg.Wait()
    close(errors)
    
    // 检查错误
    var errCount int
    for err := range errors {
        t.Errorf("并发错误: %v", err)
        errCount++
    }
    
    if errCount > 0 {
        t.Fatalf("检测到 %d 个并发错误", errCount)
    }
}

// TestRaceConditions 使用 -race 标志检测竞态条件
