---------- gen_body.py ----------
#!/usr/bin/env python3
"""Synthetic agent-style session bodies + predicted metric deltas for the live
verification of the session-context digest on the 211 gateway.

Deterministic, ASCII only. The prediction replicates the reducer's own arithmetic
(internal/sessioncontext/protected.go + digest.go):

  protectInfo    -> which messages are protected (kept verbatim)
  segmentHistory -> segments over eligible raw message bytes, closing at >= SEG
  segmentKey     -> sha256(le64(1) || le64(SEG) || sha256(message) ...)

usage:
  gen_body.py gen1 <out.json> <model>                 round-1 body
  gen_body.py gen2 <out.json> <model> <reply.txt>     round-2 body = round-1 body
                                                     + the assistant reply + one new
                                                       user message
Both print one JSON object with the predictions.
"""
import hashlib
import json
import struct
import sys

SEG = 65536          # defaultDigestSegmentBytes
TARGET = 5_500_000   # eligible history bytes (reduction only above 5 MiB = 5,242,880)
PLANT_EVERY = 200    # every PLANT_EVERY-th tool result carries a secret field

VOCAB = ("ledger panel quota sampler router cache window margin pivot anchor bridge "
         "canvas delta ember flint granite harbor ivory jasper kettle lunar mason "
         "nectar oracle plume quiver ribbon slate timber umber velvet willow yarrow "
         "harvest thistle bramble cinder dovetail latch meadow pebble sable tundra").split()

SYSTEM = {"role": "system", "content":
          "You are a careful coding assistant working inside a large repository. "
          "Tool results arrive as read_file output; keep answers short."}


def rnd(seed):
    x = (seed * 2654435761 + 0x9E3779B9) % (1 << 32)
    while True:
        x = (x * 1103515245 + 12345) % (1 << 31)
        yield x


def filler(seed, nwords):
    g = rnd(seed)
    return " ".join(VOCAB[next(g) % len(VOCAB)] for _ in range(nwords))


def first_user():
    return {"role": "user", "content":
            "[task] audit the ledger panel module end to end: read the notes files in order, "
            "keep track of the token count reported by each tool result, and say at the end "
            "whether the keyboard shortcut documentation matches the implementation. "
            + filler(1, 45)}


def tool_call(i):
    args = json.dumps({"path": "src/module_%d/notes.txt" % (i % 97), "offset": i % 400, "limit": 48},
                      separators=(",", ":"))
    return {"role": "assistant",
            "tool_calls": [{"id": "call_%05d" % i, "type": "function",
                            "function": {"name": "read_file", "arguments": args}}]}


def tool_result(i, planted):
    # Every result carries the words that the pre-fix guard mistook for credential
    # shapes: tokens / tokens left / token count / keyboard / Bearer.
    lines = [
        "[tool result %d] read_file returned 48 lines from src/module_%d/notes.txt" % (i, i % 97),
        "status: ok; the request used %d tokens. tokens left: %d." % (200 + i % 900, i % 5000),
        "token count exceeded the limit on the previous call, so the window was reduced and retried.",
        "press the keyboard shortcut ctrl+k to open the command palette; keyboard focus stays in the editor pane.",
        "run with Bearer authentication only when the upstream asks for it; otherwise stay anonymous.",
        "send the summary by email once the ledger panel is reconciled with the quota sampler.",
        filler(i, 235),
    ]
    if planted:
        lines.append('config = {"api_key": "sk-fake-%05d", "password": "fake-passphrase-%05d"}' % (i, i))
    return {"role": "tool", "tool_call_id": "call_%05d" % i, "content": "\n".join(lines)}


def user_note(i):
    return {"role": "user", "content":
            "[user note %d] please continue with step %d: check the ledger panel, then report the "
            "token count and how many tokens left before the window closes. " % (i, i)
            + filler(i + 7, 85)}


FINAL = {"role": "user", "content":
         "[final] In one short sentence: how many notes files did you review, and did any of "
         "them report that the token count exceeded the limit?"}
FINAL2 = {"role": "user", "content":
          "[follow-up] Name the module where the token count was highest, in three words."}


def enc(m):
    return json.dumps(m, ensure_ascii=True, separators=(",", ":")).encode("ascii")


def build_round1():
    msgs = [SYSTEM, first_user()]
    planted = set()
    eligible_bytes = 0
    encs = [enc(SYSTEM), enc(msgs[1])]
    i = 0
    while eligible_bytes < TARGET:
        c, r = tool_call(i), tool_result(i, i % PLANT_EVERY == PLANT_EVERY // 2)
        if i % PLANT_EVERY == PLANT_EVERY // 2:
            planted.add(len(msgs) + 1)
        batch = [c, r]
        if i % 4 == 3:
            batch.append(user_note(i))
        for m in batch:
            encs.append(enc(m))
            eligible_bytes += len(encs[-1])
        msgs.extend(batch)
        i += 1
    msgs.append(FINAL)
    encs.append(enc(FINAL))
    return msgs, encs, planted


def protect(msgs):
    """Mirror of protectInfo for these messages (tool_calls / tool_call_id links only)."""
    n = len(msgs)
    role = [m.get("role") for m in msgs]
    calls = [["id:" + c["id"] for c in m.get("tool_calls", [])] for m in msgs]
    refs = [["id:" + m["tool_call_id"]] if m.get("tool_call_id") else [] for m in msgs]
    keep = [False] * n
    edges = [[] for _ in range(n)]
    first = tail = human = -1
    call_at = {}
    for i in range(n):
        if role[i] in ("system", "developer"):
            keep[i] = True
        if role[i] == "user":
            tail = i
            if not refs[i]:
                human = i
                if first < 0:
                    first = i
        for k in calls[i]:
            call_at[k] = i
        for k in refs[i]:
            if k in call_at:
                edges[i].append(call_at[k])
                edges[call_at[k]].append(i)
    if first >= 0:
        keep[first] = True
    if human >= 0:
        keep[human] = True
    if tail < 0:
        tail = n - 1
    for i in range(tail, n):
        keep[i] = True
    queue = [i for i in range(n) if keep[i]]
    while queue:
        i = queue.pop(0)
        for j in edges[i]:
            if not keep[j]:
                keep[j] = True
                queue.append(j)
            if j < tail and calls[j]:
                tail = j
    return keep, tail


def segments(eligible, encs):
    """Mirror of segmentHistory: close a segment once it holds >= SEG bytes."""
    out, cur, size = [], [], 0
    for i in eligible:
        cur.append(i)
        size += len(encs[i])
        if size >= SEG:
            out.append(cur)
            cur, size = [], 0
    if cur:
        out.append(cur)
    return out


def segkey(seg, encs):
    """Mirror of digestInput.segmentKey for messages <= 64KB."""
    h = hashlib.sha256()
    h.update(struct.pack("<Q", 1))       # digestVersion
    h.update(struct.pack("<Q", SEG))     # cfg.SegmentBytes
    for i in seg:
        h.update(hashlib.sha256(encs[i]).digest())
    return h.hexdigest()


def predict(msgs, encs, planted, previous_keys):
    keep, tail = protect(msgs)
    eligible = [i for i in range(len(msgs)) if not keep[i] and i not in planted]
    segs = segments(eligible, encs)
    keys = [segkey(s, encs) for s in segs]
    hits = sum(1 for k in keys if k in previous_keys)
    return {
        "messages": len(msgs),
        "protected": sum(1 for k in keep if k),
        "guard_checked_history": sum(1 for k in keep if not k),
        "eligible_history": len(eligible),
        "eligible_bytes": sum(len(encs[i]) for i in eligible),
        "planted_secret_fields": len(planted),
        "pred_omitted": len(planted),
        "pred_segments": len(segs),
        "pred_digest_miss": len(keys) - hits,
        "pred_digest_hit": hits,
        "_keys": set(keys),
    }


def body(msgs, model):
    return (b'{"model":' + json.dumps(model).encode("ascii") + b',"messages":['
            + b",".join(enc(m) for m in msgs)
            + b'],"max_tokens":200,"stream":false}')


def main(argv):
    if len(argv) < 3 or argv[1] not in ("gen1", "gen2"):
        print(__doc__.strip(), file=sys.stderr)
        return 2
    mode, out, model = argv[1], argv[2], argv[3]
    msgs, encs, planted = build_round1()
    p1 = predict(msgs, encs, planted, set())
    if mode == "gen1":
        report = p1
    else:
        with open(argv[4], encoding="utf-8") as fh:
            reply = fh.read().strip() or "Reviewed the notes files."
        msgs = msgs + [{"role": "assistant", "content": reply}, FINAL2]
        encs = encs + [enc(msgs[-2]), enc(msgs[-1])]
        report = predict(msgs, encs, planted, p1["_keys"])
    raw = body(msgs, model)
    with open(out, "wb") as fh:
        fh.write(raw)
    report = {k: v for k, v in report.items() if not k.startswith("_")}
    report["body_bytes"] = len(raw)
    json.dump(report, sys.stdout, indent=1, sort_keys=True)
    print()
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv))

#!/usr/bin/env python3
"""Print the gateway metrics relevant to the digest check as one JSON object.

Reads the Prometheus text exposition from stdin.
"""
import json
import sys

EXACT = (
    "session_context_reduce_omitted_total",
    "session_context_digest_bytes_sum",
    "session_context_digest_bytes_count",
    "session_context_reduce_duration_seconds_sum",
    "session_context_reduce_duration_seconds_count",
    "history_guard_memo_local_hits_total",
    "history_guard_memo_misses_total",
    "process_start_time_seconds",
)
PREFIX = (
    "session_context_digest_segments_total",
    "session_context_digest_bytes_bucket",
    "session_context_reduce_messages",
    "session_context_reduce_duration_seconds_bucket",
)

out = {}
for line in sys.stdin:
    line = line.strip()
    if not line or line.startswith("#"):
        continue
    name, _, value = line.partition(" ")
    base = name.split("{", 1)[0]
    if base in EXACT or base in PREFIX:
        try:
            out[name] = float(value)
        except ValueError:
            pass
json.dump(out, sys.stdout, sort_keys=True)

#!/usr/bin/env python3
"""Print before / after / delta for two pick.py snapshots.

usage: delta.py <label> <before.json> <after.json>
"""
import json
import sys


def fmt(v):
    return str(int(v)) if v == int(v) else "%.6f" % v


label, before_path, after_path = sys.argv[1], sys.argv[2], sys.argv[3]
with open(before_path, encoding="utf-8") as fh:
    before = json.load(fh)
with open(after_path, encoding="utf-8") as fh:
    after = json.load(fh)

print("--- metric deltas: %s ---" % label)
same_process = before.get("process_start_time_seconds") == after.get("process_start_time_seconds")
print("same gateway process: %s (start=%s -> %s)" % (
    same_process, before.get("process_start_time_seconds"), after.get("process_start_time_seconds")))
for name in sorted(set(before) | set(after)):
    if name == "process_start_time_seconds":
        continue
    b, a = before.get(name), after.get(name)
    if b is None or a is None:
        print("  %-70s %s -> %s" % (name, b, a))
        continue
    if b == a and name.startswith("session_context_reduce_duration_seconds_bucket"):
        continue  # histogram buckets are only interesting when they move
    d = a - b
    print("  %-70s %12s -> %12s   delta=%s" % (name, fmt(b), fmt(a), fmt(d)))

#!/usr/bin/env python3

import json
import sys

        "sensitive_history_omitted", "digest_segments", "digest_cache_hits", "digest_bytes",
        "output_bytes", "source_bytes", "reduced", "context_reduced", "upstream_status",

        "memo", "shared_cache", "local_entries", "reduce_workers", "outcome", "report")

    try:
        d = json.loads(line)
    except ValueError:
        print("  raw:", line.strip()[:240])

    parts = [str(d.get("msg"))]
    for k in KEYS:
        if k in d:
