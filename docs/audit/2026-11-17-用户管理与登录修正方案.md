### 1.2 修改 `web/src/api/auth.ts`

**添加接口定义**：

```typescript
// 在文件开头添加
export interface CaptchaResponse {
  captcha_id: string
  image: string  // Base64 data URL
  expires_at: string
  required: boolean
  ttl_seconds: number
}

// 在 login 函数之前添加
export function getCaptcha() {
  return req<CaptchaResponse>('GET', '/api/auth/captcha')
}

// 修改 login 函数签名
export function login(
  username: string, 
  password: string,
  captchaId?: string,
  captchaCode?: string
) {
  const body: any = { username, password }
  if (captchaId) body.captcha_id = captchaId
  if (captchaCode) body.captcha_code = captchaCode
  return req<LoginResponse>('POST', '/api/auth/token', body)
}
```

### 1.3 修改 `web/src/components/LoginModal.vue`

**添加导入**：

**添加响应式变量**：

**修改 handleLogin 函数**：
```typescript
const handleLogin = async () => {
  if (!username.value || !password.value) {
    error.value = t('login.error.emptyFields')
    return
  }
  
  // 如果需要验证码但未填写
  if (showCaptcha.value && !captchaCode.value) {
    error.value = t('login.captcha.required')
    return
  }
  
  error.value = ''
  loading.value = true
  
  try {
    const result = await login(
      username.value, 
      password.value,
      showCaptcha.value ? captchaId.value : undefined,
      showCaptcha.value ? captchaCode.value : undefined
    )
    
    // ... 原有的成功处理逻辑 ...
    
  } catch (err: any) {
    loading.value = false
    
    // 检查是否需要验证码
    if (err.message?.includes('captcha_required') || 
        err.code === 'captcha_required') {
      showCaptcha.value = true
      error.value = t('login.captcha.requiredAfterFailures')
      // 自动刷新验证码
      setTimeout(() => captchaInputRef.value?.refresh(), 100)
      return
    }
    
    // 检查验证码错误
    if (err.message?.includes('captcha_invalid') || 
        err.code === 'captcha_invalid') {
      error.value = t('login.captcha.invalid')
      captchaCode.value = ''
      captchaInputRef.value?.refresh()
      return
    }
    
    // 其他错误
    error.value = surfaceError(err, 'login')
  }
}
```

**在表单中添加验证码组件**（在密码输入框之后）：
```vue
<CaptchaInput
  v-if="showCaptcha"
  ref="captchaInputRef"
  v-model="captchaCode"
  v-model:captcha-id="captchaId"
/>
```

## 二、认证提供者配置读取

### 2.1 修改 `cmd/gateway/main.go`

**查找认证提供者初始化代码**（约 7678 行）：

```go
// 当前代码
authRegistry := authplugin.NewRegistry(&authplugin.PluginConfig{
    AuthProvider: "native",
    // ...
})
```

**修改为**：

```go
// 从环境变量读取认证提供者配置
authProviderType := os.Getenv("LLM_GATEWAY_AUTH_PROVIDER")
if authProviderType == "" {
    authProviderType = "native" // 默认使用 native
}

pluginConfig := &authplugin.PluginConfig{
    AuthProvider: authProviderType,
    UserManager: authProviderType,      // 通常与 AuthProvider 一致
    TenantManager: authProviderType,
    ApprovalProvider: authProviderType,
}

// 如果使用 Casdoor，读取其配置
if authProviderType == "casdoor" {
    pluginConfig.CasdoorConfig = &authplugin.CasdoorConfig{
        Endpoint:     os.Getenv("LLM_GATEWAY_CASDOOR_ENDPOINT"),
        ClientID:     os.Getenv("LLM_GATEWAY_CASDOOR_CLIENT_ID"),
        ClientSecret: os.Getenv("LLM_GATEWAY_CASDOOR_CLIENT_SECRET"),
        Organization: os.Getenv("LLM_GATEWAY_CASDOOR_ORGANIZATION"),
        Application:  os.Getenv("LLM_GATEWAY_CASDOOR_APPLICATION"),
    }
    
    // 验证必需配置
    if pluginConfig.CasdoorConfig.Endpoint == "" ||
       pluginConfig.CasdoorConfig.ClientID == "" ||
       pluginConfig.CasdoorConfig.ClientSecret == "" {
        slog.Error("Casdoor configuration incomplete", 
            "missing", "CASDOOR_ENDPOINT, CLIENT_ID, or CLIENT_SECRET")
        os.Exit(1)
    }
}

authRegistry := authplugin.NewRegistry(pluginConfig)
```

### 2.2 确认 `pkg/authplugin` 中配置结构

检查 `pkg/authplugin/registry.go` 中的 `PluginConfig` 是否已包含 `CasdoorConfig`：

```go
type PluginConfig struct {
    AuthProvider     string
    UserManager      string
    TenantManager    string
    ApprovalProvider string
    
    // Provider-specific configurations
    CasdoorConfig *CasdoorConfig `json:"casdoor_config,omitempty"`
}

type CasdoorConfig struct {
    Endpoint     string `json:"endpoint"`
    ClientID     string `json:"client_id"`
    ClientSecret string `json:"client_secret"`
    Organization string `json:"organization"`
    Application  string `json:"application"`
}
```

如果不存在，需要添加这些结构体定义。

## 四、测试计划

### 4.1 验证码功能测试

**测试用例**：

1. **TC01 - 正常登录（无验证码）**
   - 输入正确用户名密码
   - 预期：直接登录成功，不显示验证码

2. **TC02 - 触发验证码**
   - 连续3次输入错误密码
   - 预期：第4次登录时显示验证码

3. **TC03 - 验证码错误**
   - 输入错误的验证码
   - 预期：提示"验证码错误"，自动刷新验证码

4. **TC04 - 验证码正确**
   - 输入正确的用户名、密码和验证码
   - 预期：登录成功

5. **TC05 - 刷新验证码**
   - 点击刷新按钮
   - 预期：显示新的验证码图片

6. **TC06 - 验证码过期**
   - 等待验证码过期（默认300秒）
   - 使用过期的验证码登录
   - 预期：提示验证码无效

### 4.2 认证提供者切换测试

**测试用例**：

1. **TC11 - 默认 Native 提供者**
   - 不设置环境变量
   - 预期：使用 Native 认证

2. **TC12 - 切换到 Casdoor**
   - 设置 `LLM_GATEWAY_AUTH_PROVIDER=casdoor`
   - 配置 Casdoor 参数
   - 重启服务
   - 预期：使用 Casdoor 认证

3. **TC13 - Casdoor 配置缺失**
   - 设置 `AUTH_PROVIDER=casdoor` 但不提供配置
   - 预期：启动失败，错误日志明确

## 八、风险与缓解

### 风险1：验证码对用户体验的影响
- **缓解**：只在多次失败后才要求验证码
- **缓解**：验证码简单易识别
- **缓解**：提供刷新功能

### 风险2：Casdoor 集成测试不足
- **缓解**：保持 Native 作为默认和回退
- **缓解**：配置验证防止启动失败
- **缓解**：详细的错误日志

### 风险3：多语言翻译质量
- **缓解**：英文和中文由开发者提供
- **缓解**：其他语种可以后续优化
- **缓解**：回退到英文不会影响功能
