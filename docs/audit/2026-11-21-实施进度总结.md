#### 2.1 修改 SessionSummaryWorker
```go
// domains/analysis/workers/session_summary_worker.go

// 添加新依赖
type SessionSummaryWorker struct {
    logger            *slog.Logger
    summarizer        SessionSummarizer
    layeredSummarizer *layeredsummary.LayeredSummarizer  // 新增
    taskAggregator    *taskaggregator.TaskAggregator     // 新增
    similarityDetector *sessiondedup.SimilarityDetector  // 新增
    closeHooks        []SessionCloseHook
    processed         atomic.Int64
    failed            atomic.Int64
}

// 修改 Handle 方法
func (w *SessionSummaryWorker) Handle(ctx context.Context, evt analysis.AnalysisEvent) error {
    // ... 现有逻辑 ...
    
    // 生成常规总结后，检查是否需要分层总结
    if w.layeredSummarizer != nil && shouldGenerateLayeredSummary(sessionKey) {
        // 判断轮次数，决定生成哪一层
        turnCount := getTurnCount(ctx, tenantID, sessionKey)
        
        if turnCount%10 == 0 && turnCount <= 50 {
            // 生成窗口总结
            w.layeredSummarizer.GenerateWindowSummary(...)
        }
        
        if turnCount%50 == 0 {
            // 生成段落总结
            windows := w.layeredSummarizer.GetLayers(ctx, tenantID, sessionKey, &layeredsummary.LayerTypeWindow)
            w.layeredSummarizer.GenerateSegmentSummary(ctx, tenantID, sessionKey, windows)
        }
    }
    
    // 查找相似会话
    if w.similarityDetector != nil && summary != nil {
        similar, _ := w.similarityDetector.FindSimilar(ctx, tenantID, sessionKey, summary.Summary, taskID)
        if len(similar) > 0 {
            similarKeys := make([]string, len(similar))
            for i, s := range similar {
                similarKeys[i] = s.SessionKey
            }
            w.similarityDetector.UpdateSimilarSessions(ctx, tenantID, sessionKey, similarKeys)
        }
    }
    
    // 更新任务聚合
    if w.taskAggregator != nil && taskID != "" {
        w.taskAggregator.UpdateTaskSummary(ctx, taskID, tenantID)
    }
    
    // ... 现有 closeHooks ...
}
```
