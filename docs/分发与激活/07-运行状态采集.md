# 07. 运行状态采集

> 通过用户授权的遥测采集，实现：
> 1. 官方对所有客户实例的运行状态监控
> 2. 不采集业务数据、用户请求、会话内容
>
> **⚠️ 采集器待实现**：`gateway/internal/collector/` 目录不存在。以下为设计目标。
> 已实现的部分仅限 `domains/hooks/observability/telemetry/`（请求级遥测记录）。
> 3. 用户可在网页中随时开关

## 一、采集范围（白名单）

### 1.1 采集 ✅

| 类别 | 字段 | 采集方式 |
|------|------|---------|
| **系统标识** | instance_id（UUID，非 IP/MAC） | 自动生成 |
| **系统标识** | version（网关版本） | version.go |
| **系统标识** | license_key_hash（SHA-256 哈希前 8 字节，16 个十六进制字符） | 客户端计算 |
| **系统资源** | uptime_secs | 启动时间差 |
| **系统资源** | cpu_usage_pct | gopsutil |
| **系统资源** | mem_used_mb / mem_total_mb | gopsutil |
| **系统资源** | disk_used_gb / disk_total_gb | gopsutil |
| **系统资源** | db_size_mb | SQL `pg_database_size()` |
| **流量指标** | current_concurrency | runtime metrics |
| **流量指标** | last_5min_tps | 5 分钟窗口计数 / 300 秒 |
| **流量指标** | last_5min_p50_ms | 5 分钟窗口 P50 |
| **流量指标** | last_5min_p99_ms | 5 分钟窗口 P99 |
| **流量指标** | last_5min_success_pct | 成功请求 / 总请求 |
| **业务指标** | model_usage（每模型请求数） | 聚合，不含内容 |
| **业务指标** | tenant_count（活跃租户数） | 聚合，仅计数 |
| **License 状态** | license_type | trial / standard / ... |
| **License 状态** | license_expires_in_days | 时间差 |

### 1.2 不采集 ❌（永远不采集）

| 类别 | 字段 |
|------|------|
| **业务数据** | prompt 内容、completion 内容、用户上传文件 |
| **会话数据** | conversation_id、message_id、session_id |
| **身份信息** | 客户邮箱（除主动提供）、API Key 明文、密码 |
| **网络身份** | IP 地址、MAC 地址、hostname（除非用户主动提供） |
| **凭证信息** | LLM 凭据（access_key、secret_key） |
| **错误明细** | 错误响应体（仅统计错误码分类） |

## 三、采集器实现

### 3.1 Collector 主循环

```go
// gateway/internal/collector/collector.go

type Collector struct {
    instanceID    string
    version       string
    licenseKey    string
    startTime     time.Time
    telemetryClient *telemetry.Client
    db            *pgxpool.Pool
    tenantStore   TenantCounter
    httpClient    *http.Client
    
    pref          *TelemetryPref  // 从 DB 读
    interval      time.Duration   // 默认 5 分钟
    reportURL     string          // llm.kxpms.cn
}

func (c *Collector) Run(ctx context.Context) {
    ticker := time.NewTicker(c.interval)
    defer ticker.Stop()
    
    for {
        select {
        case <-ctx.Done():
            return
        case <-ticker.C:
            c.collect(ctx)
        }
    }
}

func (c *Collector) collect(ctx context.Context) {
    // 0. 检查用户授权
    pref, err := c.loadPref(ctx)
    if err != nil || !pref.Enabled {
        return  // 用户禁用，跳过
    }
    
    metrics, err := c.computeMetrics(ctx)
    if err != nil {
        slog.Warn("collect failed", "error", err)
        return
    }
    
    c.report(ctx, metrics)
}
```

### 3.2 指标计算

```go
// gateway/internal/collector/metrics.go

type RuntimeMetrics struct {
    InstanceID    string `json:"instance_id"`
    Version       string `json:"version"`
    LicenseKeyHash string `json:"license_key_hash"`
    Timestamp     time.Time `json:"timestamp"`
    
    UptimeSecs   int64   `json:"uptime_secs"`
    CPUUsagePct  float64 `json:"cpu_usage_pct"`
    MemUsedMB    int64   `json:"mem_used_mb"`
    MemTotalMB   int64   `json:"mem_total_mb"`
    DiskUsedGB   int64   `json:"disk_used_gb"`
    DiskTotalGB  int64   `json:"disk_total_gb"`
    DBSizeMB     int64   `json:"db_size_mb"`
    
    CurrentConcurrency int     `json:"current_concurrency"`
    Last5MinTPS        float64 `json:"last_5min_tps"`
    Last5MinP50Ms      float64 `json:"last_5min_p50_ms"`
    Last5MinP99Ms      float64 `json:"last_5min_p99_ms"`
    Last5MinSuccessPct float64 `json:"last_5min_success_pct"`
    
    ModelUsage    map[string]int64 `json:"model_usage"`
    TenantCount   int              `json:"tenant_count"`
    
    LicenseType       string `json:"license_type"`
    LicenseExpiresIn  int    `json:"license_expires_in_days"`
}

func (c *Collector) computeMetrics(ctx context.Context) (*RuntimeMetrics, error) {
    m := &RuntimeMetrics{
        InstanceID: c.instanceID,
        Version: c.version,
        LicenseKeyHash: hashLicenseKey(c.licenseKey),
        Timestamp: time.Now(),
    }
    
    // 1. 系统资源
    if cpuPct, err := cpu.Percent(0, false); err == nil && len(cpuPct) > 0 {
        m.CPUUsagePct = cpuPct[0]
    }
    if mem, err := mem.VirtualMemory(); err == nil {
        m.MemUsedMB = int64(mem.Used) / 1024 / 1024
        m.MemTotalMB = int64(mem.Total) / 1024 / 1024
    }
    if disk, err := disk.Usage("/"); err == nil {
        m.DiskUsedGB = int64(disk.Used) / 1024 / 1024 / 1024
        m.DiskTotalGB = int64(disk.Total) / 1024 / 1024 / 1024
    }
    
    // 2. DB 大小（不读内容）
    if dbSize, err := queryDBSize(ctx, c.db); err == nil {
        m.DBSizeMB = dbSize
    }
    
    // 3. 流量指标
    m.Last5MinTPS = c.telemetryClient.GetTPS(5 * time.Minute)
    m.Last5MinP50Ms = c.telemetryClient.GetLatencyP50(5 * time.Minute)
    m.Last5MinP99Ms = c.telemetryClient.GetLatencyP99(5 * time.Minute)
    m.Last5MinSuccessPct = c.telemetryClient.GetSuccessRate(5 * time.Minute)
    m.CurrentConcurrency = c.telemetryClient.GetCurrentConcurrency()
    
    // 4. 模型使用统计（聚合）
    m.ModelUsage = c.telemetryClient.GetModelUsageStats(5 * time.Minute)
    
    // 5. 租户数（仅计数）
    if count, err := c.tenantStore.CountActive(ctx); err == nil {
        m.TenantCount = count
    }
    
    // 6. uptime & license
    m.UptimeSecs = int64(time.Since(c.startTime).Seconds())
    m.LicenseType, m.LicenseExpiresIn = c.getLicenseInfo()
    
    return m, nil
}

func hashLicenseKey(key string) string {
    h := sha256.Sum256([]byte(key))
    return hex.EncodeToString(h[:8])  // 仅 16 字符
}
```

### 3.3 上报到主控端

```go
// gateway/internal/collector/reporter.go

func (c *Collector) report(ctx context.Context, m *RuntimeMetrics) error {
    payload, err := json.Marshal(m)
    if err != nil {
        return err
    }
    
    req, err := http.NewRequestWithContext(ctx, "POST",
        "https://llm.kxpms.cn/api/v1/collect/runtime",
        bytes.NewReader(payload))
    if err != nil {
        return err
    }
    
    req.Header.Set("Content-Type", "application/json")
    req.Header.Set("X-Instance-ID", c.instanceID)
    req.Header.Set("Authorization", "Bearer "+c.collectToken)
    
    resp, err := c.httpClient.Do(req)
    if err != nil {
        // 上报失败不阻塞主流程
        slog.Warn("telemetry report failed", "error", err)
        return nil
    }
    defer resp.Body.Close()
    
    if resp.StatusCode != 200 {
        slog.Warn("telemetry report non-200", "status", resp.StatusCode)
    }
    return nil
}
```

### 4.1 主控端 API：/api/v1/collect/runtime

```go
// license-authority/api/collect.go

func HandleCollectRuntime(c *gin.Context) {
    var m RuntimeMetrics
    if err := c.Bind(&m); err != nil {
        return error(err)
    }
    
    // 1. 验证 instance_id
    if err := verifyInstance(m.InstanceID); err != nil {
        return error(err)
    }
    
    // 2. 写入时序库（PostgreSQL）
    if err := store.InsertRuntimeMetrics(c.Request().Context(), m); err != nil {
        return error(err)
    }
    
    // 3. 异常检测
    if m.CPUUsagePct > 95 {
        alertManager.Fire(Alert{
            Type: "cpu_high",
            InstanceID: m.InstanceID,
            Severity: "warning",
        })
    }
    if float64(m.DiskUsedGB) / float64(m.DiskTotalGB) > 0.95 {
        alertManager.Fire(Alert{
            Type: "disk_high",
            InstanceID: m.InstanceID,
            Severity: "critical",
        })
    }
    
    return c.JSON(200, gin.H{"status": "ok"})
}
```

// gateway/internal/api/telemetry_handler.go

func HandleGetPref(c *gin.Context) {
    pref, err := db.GetTelemetryPref(c.Request().Context())
    if err != nil {
        pref = &TelemetryPref{Enabled: false, IntervalSec: 300}
    }
    return c.JSON(200, pref)
}

func HandleUpdatePref(c *gin.Context) {
    var req struct {
        Enabled     bool `json:"enabled"`
        IntervalSec int  `json:"interval_sec"`
    }
    if err := c.Bind(&req); err != nil {
        return error(err)
    }
    
    if err := db.UpsertTelemetryPref(c.Request().Context(), req); err != nil {
        return error(err)
    }
    
    return c.JSON(200, gin.H{"status": "saved"})
}

## 六、数据隔离与安全

| 维度 | 措施 |
|------|------|
| **传输加密** | TLS 1.3（强加密套件） |
| **license_key 不外泄** | 仅发送 SHA-256 前 16 字符 |
| **IP/MAC/hostname 不外泄** | 仅发送 instance_id（UUID） |
| **业务数据不采集** | 客户端代码不读取 prompt/completion 文件 |
| **可逆** | 用户随时关闭，关闭后立即停止上报 |

## 七、关键决策

| 决策 | 选择 | 理由 |
|------|------|------|
| 默认启用？ | ❌ 默认关闭 | GDPR / 合规优先 |
| 询问时机 | 首次激活时 | 一次性，提升激活率 |
| 采集频率 | 默认 5 分钟 | 平衡开销与实时性 |
| 关闭后是否删除已采集数据 | 是（下次连接时附带删除请求） | 隐私尊重 |
