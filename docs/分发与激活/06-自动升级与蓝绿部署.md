# 06. 自动升级与蓝绿部署

> 客户端升级通过独立的升级守护进程 + 蓝绿切换实现零停机可回退升级。
>
> **⚠️ 全部待实现**：本模块的代码（`upgrader/`、`gateway/internal/api/upgrade_handler.go`、
> `scripts/kx-gateway-rollback`）在当前代码库中均不存在。以下为设计目标。
> 已实现的部分仅限 `autoupdate/*.go`（Downloader/Installer/Rollback）。

### 4.3 升级编排核心

```go
// upgrader/internal/upgrader.go

func (u *Upgrader) Run(targetVersion string, cb Callbacks) error {
    defer u.persistState()
    
    // Phase 1: 检查更新
    u.toState(CHECKING, cb)
    manifest, err := u.fetchManifest(targetVersion)
    if err != nil { return err }
    
    // Phase 2: 下载（不破坏原版本）
    u.toState(DOWNLOADING, cb)
    if err := u.download(manifest); err != nil { return err }
    
    // Phase 3: 在 green slot 安装（独立 slot）
    u.toState(INSTALLING, cb)
    if err := u.installToGreenSlot(manifest); err != nil { return err }
    
    // Phase 4: 测试新版本
    u.toState(TESTING, cb)
    if err := u.testGreenSlot(); err != nil {
        u.cleanupGreenSlot()
        u.toState(FAILED, cb)
        return err
    }
    
    // Phase 5: 切换主入口
    u.toState(SWITCHING, cb)
    if err := u.switchMainToGreen(); err != nil { return err }
    
    // Phase 6: 监控期（30 分钟可回退）
    u.toState(MONITORING, cb)
    if err := u.monitorWithGrace(DefaultRollbackGrace); err != nil {
        u.Rollback()
        return err
    }
    
    // Phase 7: 清理 blue slot
    u.toState(STOPPING_BLUE, cb)
    if err := u.stopBlueSlot(); err != nil { return err }
    
    // Phase 8: 完成
    u.toState(IDLE, cb)
    return nil
}
```

## 五、蓝绿切换实现

### 5.1 nginx upstream 方式（推荐）

```nginx
# /etc/nginx/conf.d/kx-gateway.conf

upstream kx_gateway {
    server 127.0.0.1:8782;  # 当前 blue slot
    keepalive 32;
}

server {
    listen 8781;
    server_name _;
    
    location / {
        proxy_pass http://kx_gateway;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }
    
    location /healthz {
        proxy_pass http://kx_gateway/healthz;
    }
}
```

切换通过修改 upstream server 端口实现（毫秒级切换）：

```go
// 切换主入口到 green
func (u *Upgrader) switchMainToGreen() error {
    conf := fmt.Sprintf(`upstream kx_gateway {
        server 127.0.0.1:%d;  # green = 8783
        keepalive 32;
    }`, GreenSlotPort)
    
    // 1. 写入新配置
    if err := os.WriteFile(NginxConfPath, []byte(conf), 0644); err != nil {
        return err
    }
    
    // 2. 测试配置语法
    if err := exec.Command("nginx", "-t").Run(); err != nil {
        return fmt.Errorf("nginx config test failed: %w", err)
    }
    
    // 3. 重新加载（不中断连接）
    return exec.Command("nginx", "-s", "reload").Run()
}
```

### 5.2 iptables 方式（兜底）

```bash
# 切换主入口 8781 → 8783 (green)
iptables -t nat -F PREROUTING  # 清空旧规则
iptables -t nat -A PREROUTING -p tcp --dport 8781 \
    -j REDIRECT --to-port 8783
```

切换时间 < 50ms。

### 5.3 systemd socket activation（高级）

```go
// 通过切换 systemd unit 指向 green slot
func (u *Upgrader) switchViaSystemd() error {
    unitPath := "/etc/systemd/system/kx-gateway-main.service"
    content := fmt.Sprintf(`[Service]
Environment=ACTIVE_SLOT=green
ExecStart=/usr/bin/docker compose -f %s/compose.yml up
ExecStop=/usr/bin/docker compose -f %s/compose.yml down
`, GreenSlotDir, GreenSlotDir)
    
    os.WriteFile(unitPath, []byte(content), 0644)
    exec.Command("systemctl", "daemon-reload").Run()
    return exec.Command("systemctl", "restart", "kx-gateway-main").Run()
}
```

## 六、独立回退 CLI

```bash
#!/bin/bash
# scripts/kx-gateway-rollback
# 独立可执行，任何状态都可回退

set -euo pipefail

DATA_ROOT="${KX_GATEWAY_DATA_ROOT:-/var/lib/kx-gateway}"
STATE_FILE="$DATA_ROOT/upgrader.state.json"

echo "╔══════════════════════════════════════════╗"
echo "║  KX Gateway 紧急回退工具                 ║"
echo "╚══════════════════════════════════════════╝"

if [[ ! -f "$STATE_FILE" ]]; then
    echo "❌ 状态文件不存在，无法自动回退"
    echo "手动回退："
    echo "  1. 检查哪个 slot 当前健康："
    echo "     curl -sf http://localhost:8782/healthz  # blue"
    echo "     curl -sf http://localhost:8783/healthz  # green"
    echo "  2. 修改 nginx 配置指向健康 slot"
    echo "  3. nginx -s reload"
    exit 1
fi

LAST_GOOD=$(jq -r '.last_good_slot' "$STATE_FILE")
LAST_GOOD_PORT=$([ "$LAST_GOOD" == "blue" ] && echo 8782 || echo 8783)

echo "  最后已知良好 slot: $LAST_GOOD (port $LAST_GOOD_PORT)"
echo ""

# 1. 检测目标 slot 健康
echo "▶ 检测 $LAST_GOOD slot 健康状态..."
if curl -sf "http://localhost:$LAST_GOOD_PORT/healthz" > /dev/null; then
    echo "  ✅ 健康"
else
    echo "  ⚠️  不健康，但仍尝试切换"
    read -p "  继续? (y/N): " -n 1 -r
    echo
    [[ ! $REPLY =~ ^[Yy]$ ]] && exit 1
fi

# 2. 切换主入口
echo "▶ 切换主入口..."
case "${KX_SWITCH_METHOD:-nginx}" in
    nginx)
        sed -i "s/127\.0\.0\.1:878[23]/127.0.0.1:$LAST_GOOD_PORT/" \
            /etc/nginx/conf.d/kx-gateway.conf
        nginx -s reload
        ;;
    iptables)
        iptables -t nat -F PREROUTING
        iptables -t nat -A PREROUTING -p tcp --dport 8781 \
            -j REDIRECT --to-port $LAST_GOOD_PORT
        ;;
esac

# 3. 停止故障 slot
BAD_SLOT=$([ "$LAST_GOOD" == "blue" ] && echo "green" || echo "blue")
if [[ -d "$DATA_ROOT/slot-$BAD_SLOT" ]]; then
    echo "▶ 停止 $BAD_SLOT slot..."
    cd "$DATA_ROOT/slot-$BAD_SLOT"
    docker compose down 2>/dev/null || true
fi

# 4. 健康验证
sleep 5
echo "▶ 健康检查..."
if curl -sf http://localhost:8781/healthz > /dev/null; then
    echo "✅ 回退成功（用时 $(($SECONDS)) 秒）"
    echo "[$(date)] ROLLBACK from $BAD_SLOT to $LAST_GOOD" \
        >> "$DATA_ROOT/audit.log"
    exit 0
else
    echo "❌ 回退失败，需要人工介入"
    exit 1
fi
```
