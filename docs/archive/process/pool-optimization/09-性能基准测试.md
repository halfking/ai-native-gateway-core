```go
// TestScheduler_MemoryLeak 测试内存泄漏
func TestScheduler_MemoryLeak(t *testing.T) {
    candidates := makeMockCandidates(100)
    scheduler := NewScheduler(NewWRRPipeline())

    ctx := context.Background()

    // 记录初始内存
    var m1 runtime.MemStats
    runtime.GC()
    runtime.ReadMemStats(&m1)

    // 执行 1M 次选路
    for i := 0; i < 1_000_000; i++ {
        scheduler.Select(ctx, candidates)
    }

    // 记录结束内存
    var m2 runtime.MemStats
    runtime.GC()
    runtime.ReadMemStats(&m2)

    // 内存增量应 <50MB
    increase := m2.Alloc - m1.Alloc
    assert.Less(t, increase, uint64(50*1024*1024), "memory leak detected")
}
```

```python
import json
import sys

def parse_report(filename):
    with open(filename) as f:
        data = json.load(f)
    return {
        'p50': data['latencies']['p50'],
        'p99': data['latencies']['p99'],
        'throughput': data['throughput'],
        'success_rate': data['success_ratio']
    }

baseline = parse_report(sys.argv[1])
wrr = parse_report(sys.argv[2])
hybrid = parse_report(sys.argv[3])

print("| Metric       | Baseline | WRR      | Hybrid   | WRR vs Baseline | Hybrid vs Baseline |")
print("|--------------|----------|----------|----------|-----------------|-------------------|")

for metric in ['p50', 'p99', 'throughput', 'success_rate']:
    b = baseline[metric]
    w = wrr[metric]
    h = hybrid[metric]

    w_diff = (w - b) / b * 100
    h_diff = (h - b) / b * 100

    print(f"| {metric:12} | {b:8.2f} | {w:8.2f} | {h:8.2f} | {w_diff:+6.1f}%      | {h_diff:+6.1f}%        |")
```

```yaml
      - name: Compare

          go install golang.org/x/perf/cmd/benchstat@latest
          benchstat baseline.txt new.txt

      - name: Check regression
        run: |
          python3 .github/scripts/check-benchmark-regression.py baseline.txt new.txt
```

**回归检查脚本**：`.github/scripts/check-benchmark-regression.py`

```python
#!/usr/bin/env python3
import sys
import re

def parse_benchmark(line):
    # BenchmarkSmoothWRR_Select/N=100-8  300000  4200 ns/op
    match = re.match(r'(\w+)\s+\d+\s+(\d+)\s+ns/op', line)
    if match:
        return match.group(1), int(match.group(2))
    return None, None

baseline = {}
with open(sys.argv[1]) as f:
    for line in f:
        name, ns = parse_benchmark(line)
        if name:
            baseline[name] = ns

new = {}
with open(sys.argv[2]) as f:
    for line in f:
        name, ns = parse_benchmark(line)
        if name:
            new[name] = ns

# 检查回归
failed = False
for name, old_ns in baseline.items():
    new_ns = new.get(name)
    if new_ns is None:
        continue

    regression = (new_ns - old_ns) / old_ns * 100

    if regression > 10:  # 允许 10% 性能退化
        print(f"❌ REGRESSION: {name}: {old_ns}ns → {new_ns}ns ({regression:+.1f}%)")
        failed = True
    else:
        print(f"✅ OK: {name}: {old_ns}ns → {new_ns}ns ({regression:+.1f}%)")

sys.exit(1 if failed else 0)
```
