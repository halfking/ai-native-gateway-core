// SmoothWRR 实现 Nginx 平滑加权轮询
type SmoothWRR struct {
    candidates []*wrrItem
    totalWeight int
    mu          sync.Mutex
}

type wrrItem struct {
    candidate     provider.Candidate
    weight        int     // 静态配置权重
    currentWeight int     // 动态平滑权重
}

// Select 按 Smooth WRR 选出一个候选
func (w *SmoothWRR) Select() *provider.Candidate {
    w.mu.Lock()
    defer w.mu.Unlock()

    if len(w.candidates) == 0 {
        return nil
    }

    var best *wrrItem
    for _, item := range w.candidates {
        item.currentWeight += item.weight
        if best == nil || item.currentWeight > best.currentWeight {
            best = item
        }
    }

    best.currentWeight -= w.totalWeight
    return &best.candidate
}

### 2.3 权重自动计算

```go
// domains/scheduler/weight_calculator.go

type WeightCalculator struct {
    // RPM 维度权重占比
    rpmWeightFactor    float64 // 0.6
    // TPM 维度权重占比
    tpmWeightFactor    float64 // 0.3
    // 历史成功率的加成
    successRateFactor  float64 // 0.1
}

// Calculate 根据凭据的 RPM/TPM 和历史表现计算权重
func (wc *WeightCalculator) Calculate(c *provider.Candidate) int {
    rpm := derefInt(c.RPMLimit)
    tpm := derefTPMLimit(c.TPMLimit)

    // 基础权重：RPM 占主导
    baseWeight := float64(rpm) * wc.rpmWeightFactor

    // TPM 加成
    tpmWeight := float64(tpm) / 1000 * wc.tpmWeightFactor

    // 历史成功率加成
    successBonus := c.SuccessRate * 100 * wc.successRateFactor

    return int(baseWeight + tpmWeight + successBonus)
}

// Normalize 归一化权重到 1-100 范围
func (wc *WeightCalculator) Normalize(weights []int) []int {
    maxW := max(weights...)
    if maxW <= 0 { return weights }

    normalized := make([]int, len(weights))
    for i, w := range weights {
        normalized[i] = max(1, w*100/maxW)
    }
    return normalized
}
```

### 2.4 号池水位监控

```go
// domains/watermark/monitor.go

type PoolWatermark struct {
    TotalAvailableRPM     int     // 号池总可用 RPM
    TotalAvailableTPM     int64   // 号池总可用 TPM
    ActiveCredentials     int     // 活跃凭据数
    HealthyCredentials    int     // 健康凭据数
    DegradedCredentials   int     // 降级凭据数
    CircuitOpen           int     // 熔断中凭据数
    SafetyLine            int     // 安全线 (RPM)
    UsageRate             float64 // 使用率 (0-1)
    IsHealthy             bool    // 是否健康
}

type WatermarkMonitor struct {
    db          *pgxpool.Pool
    redis       *redis.Client
    alertChan   chan<- Alert

    safetyMargin float64 // 安全余量, 默认 1.5
}

// Check 执行一次水位检查
func (m *WatermarkMonitor) Check(ctx context.Context) (*PoolWatermark, error) {
    // 1. 从 DB 统计凭据状态
    stats, _ := m.loadPoolStats(ctx)

    // 2. 从 Redis 获取实时滑动窗口用量
    usage, _ := m.loadRealtimeUsage(ctx)

    // 3. 计算可用吞吐量
    available := m.calcAvailableThroughput(stats, usage)

    // 4. 计算安全线
    safetyLine := int(float64(available.TotalLimit) / m.safetyMargin)

    // 5. 判断水位并触发告警
    watermark := &PoolWatermark{
        TotalAvailableRPM:  available.TotalLimit - usage.CurrentRPM,
        TotalAvailableTPM:  available.TotalTPMLimit - usage.CurrentTPM,
        UsageRate:          float64(usage.CurrentRPM) / float64(available.TotalLimit),
        SafetyLine:         safetyLine,
        IsHealthy:          available.TotalLimit - usage.CurrentRPM >= safetyLine,
    }

    m.evaluateAndAlert(watermark)
    return watermark, nil
}

func (m *WatermarkMonitor) evaluateAndAlert(watermark *PoolWatermark) {
    switch {
    case watermark.TotalAvailableRPM <= 0:
        m.alertChan <- Alert{Level: Emergency, Msg: "号池耗尽! 无可用凭据"}
    case watermark.UsageRate > 0.85:
        m.alertChan <- Alert{Level: Critical, Msg: "号池吞吐量不足, 使用率 >85%"}
    case watermark.UsageRate > 0.70:
        m.alertChan <- Alert{Level: Warn, Msg: "号池水位偏高, 使用率 >70%"}
    }
}
```

### 2.5 并发排队缓冲

```go
// domains/credential/limiter.go 新增排队机制

// AcquireQueue 有界优先级队列等待
func (l *Limiter) AcquireQueue(ctx context.Context, providerID, credentialID int, priority int) (ReleaseFunc, error) {
    // 1. 先尝试非阻塞获取
    if l.tryAcquireLight(providerID, credentialID) {
        return l.buildReleaseFunc(...)
    }

    // 2. 进入优先级队列
    entry := &queueEntry{
        credentialID: credentialID,
        priority:     priority,
        ch:           make(chan struct{}),
    }
    l.queue.Push(entry)

    // 3. 等待出队或超时
    select {
    case <-entry.ch:
        return l.buildReleaseFunc(...)
    case <-ctx.Done():
        l.queue.Remove(entry)
        return nil, ctx.Err()
    case <-time.After(5 * time.Second):
        l.queue.Remove(entry)
        return nil, ErrQueueTimeout
    }
}

## 5. 与现有系统兼容性

| 现有特性 | 影响 | 兼容策略 |
|---|---|---|
| Bandit scorer | 新增 WRR 策略后 Bandit 保留 | 新增策略枚举，默认维持 Bandit |
| 4 层限流器 | 增加排队缓冲 | 排队不改变现有信号量语义 |
| 熔断器 | 无影响 | 选择器在 candidate pool 层已过滤熔断 |
| 指纹槽 | 无影响 | WRR 在槽分配之前，互不干扰 |
| 健康检查 | 无影响 | 独立运行，输出 degraded 状态给调度器 |
| 后台恢复 | 无影响 | 恢复后权重逐步恢复 |

## 6. 配置迁移路径

```
Phase 1 (默认):  Bandit (现有行为, 零迁移)
Phase 2 (可选):  Smooth WRR + 自动权重计算 (新项目启用)
Phase 3 (推荐):  Bandit + WRR 混合 (Bandit 主选, WRR 兜底)
Phase 4 (高级):  Composite Scoring (多目标优化)
