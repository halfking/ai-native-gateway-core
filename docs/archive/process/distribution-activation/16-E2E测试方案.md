### 3.3 通用测试函数

```bash
# tests/lib/common.sh

# 运行单个测试
run_test() {
    local test_name="$1"
    local test_script="$2"
    
    echo "========================================="
    echo "Running test: $test_name"
    echo "========================================="
    
    start_time=$(date +%s)
    
    if bash "$test_script"; then
        end_time=$(date +%s)
        duration=$((end_time - start_time))
        record_test_result "$test_name" "PASS" "$duration"
        echo "✅ Test passed: $test_name (${duration}s)"
    else
        end_time=$(date +%s)
        duration=$((end_time - start_time))
        record_test_result "$test_name" "FAIL" "$duration"
        echo "❌ Test failed: $test_name (${duration}s)"
        return 1
    fi
}

# 等待服务就绪
wait_for_service() {
    local url="$1"
    local max_retries="${2:-30}"
    local retry_interval="${3:-2}"
    
    for i in $(seq 1 $max_retries); do
        if curl -fsS "$url" > /dev/null 2>&1; then
            return 0
        fi
        sleep $retry_interval
    done
    
    return 1
}

# 清理测试环境
cleanup_test_env() {
    echo "Cleaning up test environment..."
    
    # 停止服务
    systemctl stop llm-gateway-go || true
    
    # 清理数据库
    sudo -u postgres psql -c "DROP DATABASE IF EXISTS llm_gateway_test;"
    
    # 清理文件
    rm -rf /var/lib/kx-gateway-test
    
    echo "✅ Cleanup completed"
}
```

### 3.4 断言函数

```bash
# tests/lib/assertions.sh

assert_equals() {
    local expected="$1"
    local actual="$2"
    local message="${3:-Assertion failed}"
    
    if [[ "$expected" == "$actual" ]]; then
        echo "✅ $message"
        return 0
    else
        echo "❌ $message"
        echo "   Expected: $expected"
        echo "   Actual:   $actual"
        return 1
    fi
}

assert_contains() {
    local haystack="$1"
    local needle="$2"
    local message="${3:-String not found}"
    
    if echo "$haystack" | grep -q "$needle"; then
        echo "✅ $message"
        return 0
    else
        echo "❌ $message"
        echo "   Looking for: $needle"
        echo "   In: $haystack"
        return 1
    fi
}

assert_http_status() {
    local url="$1"
    local expected_status="$2"
    local message="${3:-HTTP status check failed}"
    
    local actual_status=$(curl -s -o /dev/null -w "%{http_code}" "$url")
    
    assert_equals "$expected_status" "$actual_status" "$message"
}

assert_service_running() {
    local service="$1"
    local message="${2:-Service not running}"
    
    if systemctl is-active --quiet "$service"; then
        echo "✅ $message"
        return 0
    else
        echo "❌ $message"
        return 1
    fi
}
```

## 四、CI/CD 集成

### 4.1 GitHub Actions 工作流

```yaml
# .github/workflows/e2e-test.yml
name: E2E Tests

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]
  workflow_dispatch:
    inputs:
      test_suite:
        description: 'Test suite to run'
        required: true
        default: 'all'
        type: choice
        options:
          - all
          - install
          - upgrade
          - docker

jobs:
  e2e-test:
    runs-on: ubuntu-latest
    
    steps:
      - name: Checkout code
        uses: actions/checkout@v3
      
      - name: Set up Go
        uses: actions/setup-go@v4
        with:
          go-version: '1.21'
      
      - name: Build
        run: |
          make build
      
      - name: Run E2E tests
        run: |
          bash tests/run-e2e-suite.sh local ${{ github.event.inputs.test_suite || 'all' }}
      
      - name: Upload test report
        if: always()
        uses: actions/upload-artifact@v3
        with:
          name: e2e-test-report
          path: tests/reports/
      
      - name: Notify on failure
        if: failure()
        run: |
          curl -X POST ${{ secrets.WEBHOOK_URL }} \
            -H "Content-Type: application/json" \
            -d '{"text": "E2E tests failed on ${{ github.ref }}"}'
```
