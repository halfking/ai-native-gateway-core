```go
// 伪代码：OmniFree资源请求
func HandleAutoRequest(ctx context.Context, req *Request) (*Response, error) {
    // Step 1: 解析 model=auto
    if req.Model == "auto" {
        tiers := []string{"subscription", "api", "cheap", "free"}
    }

    // Step 2: 遍历tier候选
    for _, tier := range tiers {
        if tier == "free" {
            // Step 3: 查询免费资源目录
            resources := freeResourceCatalog.Query(ctx, req.Tenant)

            for _, res := range resources {
                // Step 4: 检查quota状态
                quotaState := quotaTracker.Check(ctx, res.ProviderID, res.ModelID)

                if quotaState == "available" {
                    // Step 5: 多Key轮换选择
                    key, err := multiKeyRotator.SelectKey(ctx, res.CredentialID)
                    if err != nil {
                        continue
                    }

                    // Step 6: 执行请求
                    resp, err := executeUpstream(ctx, res, key, req)
                    if err != nil {
                        // Step 7: 429错误精细分类
                        errType := classify429(err)
                        switch errType {
                        case "rate_limit":
                            // 等待Retry-After，不切换资源
                            waitRetryAfter(err)
                            continue
                        case "quota_exhausted":
                            // 标记资源耗尽，切换下一个
                            quotaTracker.MarkExhausted(ctx, res)
                            continue
                        }
                    }

                    // Step 8: 成功，更新quota追踪
                    quotaTracker.RecordUsage(ctx, res, resp.Usage)
                    return resp, nil
                }
            }
        }
    }

    return nil, ErrAllTiersExhausted
}
```

#### 多Key轮换逻辑

```go
type MultiKeyRotator struct {
    strategy string // "round_robin" | "least_recent_error"
}

func (r *MultiKeyRotator) SelectKey(ctx context.Context, credID int) (*Key, error) {
    keys := db.Query("SELECT * FROM credential_keys WHERE credential_id = $1 AND status = 'active'", credID)

    // 过滤冷却中的key
    available := filterNotInCooldown(keys)
    if len(available) == 0 {
        return nil, ErrAllKeysInCooldown
    }

    switch r.strategy {
    case "round_robin":
        return available[0], nil  // 按key_index排序取第一个
    case "least_recent_error":
        sort.Slice(available, func(i, j int) bool {
            return available[i].LastErrorAt.Before(available[j].LastErrorAt)
        })
        return available[0], nil
    }
}

func (r *MultiKeyRotator) RecordError(ctx context.Context, keyID int, err error) {
    // 增加错误计数
    db.Exec("UPDATE credential_keys SET error_count = error_count + 1, last_error_at = NOW() WHERE id = $1", keyID)

    // 触发冷却
    if errorCount >= 3 {
        cooldownDuration := parseCooldownFromError(err)  // 例如 "Retry-After: 60s"
        db.Exec("UPDATE credential_keys SET status = 'cooling', cooldown_until = NOW() + $1 WHERE id = $2", cooldownDuration, keyID)
    }
}
```

### 1.4 429错误精细分类

```go
func Classify429(err error, respBody string) string {
    // 优先级1: HTTP header
    if retryAfter := err.Response.Header.Get("Retry-After"); retryAfter != "" {
        // 解析相对单位: "6s", "5m", "2h", "1d"
        duration := parseRelativeRetryAfter(retryAfter)
        if duration < 5*time.Minute {
            return "rate_limit"  // 短期限流
        }
        return "quota_exhausted"  // 长期耗尽
    }

    // 优先级2: body关键词
    bodyLower := strings.ToLower(respBody)
    if strings.Contains(bodyLower, "rate limit") || strings.Contains(bodyLower, "too many requests") {
        return "rate_limit"
    }
    if strings.Contains(bodyLower, "quota") || strings.Contains(bodyLower, "credit") || strings.Contains(bodyLower, "balance") {
        return "quota_exhausted"
    }

    // 优先级3: provider特定规则
    switch provider {
    case "groq":
        // Groq用"x-ratelimit-remaining-requests: 0"表示quota耗尽
        if err.Response.Header.Get("x-ratelimit-remaining-requests") == "0" {
            return "quota_exhausted"
        }
    case "deepseek":
        // DeepSeek在body包含"insufficient_quota"
        if strings.Contains(bodyLower, "insufficient_quota") {
            return "quota_exhausted"
        }
    }

    return "rate_limit"  // 默认保守处理为rate_limit
}
```

### 2.2 压缩器工作流

```go
func (sc *SessionCompressor) Prepare(ctx context.Context, req *Request) (*CompressedSession, error) {
    // Phase 0: session_id校验
    if req.SessionID == "" {
        req.SessionID = generateSessionID()
    }

    // Phase 1: 读取状态 (V2优先)
    state, err := sc.readState(ctx, req.SessionID)
    if err == ErrStateNotFound {
        // 首次请求，初始化空状态
        state = &SessionState{SchemaVersion: "v2"}
    }

    // Phase 2: delta-append (增量消息)
    newMessages := req.Messages[len(state.Messages):]  // 只处理新增消息
    state.Messages = append(state.Messages, newMessages...)
    state.MsgCount += len(newMessages)
    state.TokenEstimate += estimateTokens(newMessages)

    // Phase 3: tools缓存
    if req.Tools != nil {
        toolsHash := hashTools(req.Tools)
        if toolsHash != state.ToolsHash {
            // Tools变更，缓存新版本
            state.Tools = req.Tools
            state.ToolsHash = toolsHash
        }
    }

    // Phase 4: 滑动窗口评估
    if shouldTriggerCompression(state) {
        // 触发条件：
        // - MsgCount > 20
        // - TokenEstimate > 8000
        // - 距离上次压缩 > 10min

        compressed, err := sc.compress(ctx, state)
        if err != nil {
            log.Warn("compression failed, continue without", "err", err)
        } else {
            state = compressed
            state.LastCompressedAt = time.Now()
        }
    }

    // Phase 5: 写回缓存
    sc.updateCache(ctx, req.SessionID, state)

    return &CompressedSession{
        SessionID: req.SessionID,
        Messages:  state.Messages,
        Tools:     state.Tools,
    }, nil
}

func (sc *SessionCompressor) compress(ctx context.Context, state *SessionState) (*SessionState, error) {
    // 压缩策略1: 生成摘要
    if state.MsgCount > 20 {
        summary, err := sc.generateSummary(ctx, state.Messages[:10])  // 前10轮摘要
        if err == nil {
            state.SummaryMarker = 10
            state.Messages = append(
                []Message{{Role: "system", Content: "Previous conversation summary: " + summary}},
                state.Messages[10:]...,  // 保留后面的轮次
            )
        }
    }

    // 压缩策略2: 裁剪过期消息
    if state.TokenEstimate > 10000 {
        // 保留最近15轮 + 摘要
        if state.SummaryMarker > 0 {
            state.Messages = state.Messages[len(state.Messages)-15:]
        }
    }

    return state, nil
}
```

```go
func (ir *IRAdapter) ConvertMultimodal(ctx context.Context, req *MultimodalRequest) (*ProviderRequest, error) {
    // Step 1: tier-based推断
    tier := inferModalityTier(req.Provider, req.Contents)

    case 1:  // 原生支持
        return ir.convertNative(ctx, req)

    case 2:  // 需要转换

    case 3:  // 降级
        return ir.degradeToText(ctx, req)
    }

func (ir *IRAdapter) convertNative(ctx context.Context, req *MultimodalRequest) (*ProviderRequest, error) {
    switch req.Provider {
    case "gemini":
        // Gemini特殊处理: CSV文档需要base64-encode
        for i, content := range req.Contents {
            if content.Type == "document" && content.MimeType == "text/csv" {
                content.Data = base64.StdEncoding.EncodeToString([]byte(content.Data))
                content.Encoding = "base64"
                req.Contents[i] = content
            }
        }
        return &ProviderRequest{
            Model: req.Model,
            Contents: ir.toGeminiContents(req.Contents),
        }, nil

    case "claude":
        // Claude: 多个content blocks
        return &ProviderRequest{
            Model: req.Model,
            Messages: []Message{
                {
                    Role: "user",
                    Content: ir.toClaudeContentBlocks(req.Contents),
                },
            },
        }, nil
    }
}

func (ir *IRAdapter) convertWithTransformation(ctx context.Context, req *MultimodalRequest) (*ProviderRequest, error) {
    switch req.Provider {
    case "openai":
        // OpenAI: image_url vs base64
        contents := []Content{}
        for _, c := range req.Contents {
            if c.Type == "image" {
                if isPublicURL(c.Source) {
                    contents = append(contents, Content{
                        Type: "image_url",
                        ImageURL: &ImageURL{URL: c.Source},
                    })
                } else {
                    // 本地文件 → base64
                    data := base64.StdEncoding.EncodeToString(c.Data)
                    contents = append(contents, Content{
                        Type: "image_url",
                        ImageURL: &ImageURL{
                            URL: fmt.Sprintf("data:%s;base64,%s", c.MimeType, data),
                        },
                    })
                }
            } else {
                contents = append(contents, Content{Type: "text", Text: c.Text})
            }
        }
        return &ProviderRequest{
            Model: req.Model,
            Messages: []Message{{Role: "user", Content: contents}},
        }, nil
    }
}
```

### 3.3 IR Circuit Breaker (Per-Provider隔离)

```go
type IRCircuitBreaker struct {
    breakers map[string]*CircuitBreaker  // key: provider
}

func (cb *IRCircuitBreaker) Execute(ctx context.Context, provider string, fn func() error) error {
    breaker := cb.getOrCreateBreaker(provider)

    return breaker.Call(func() error {
        err := fn()
        if err != nil {
            // 分类错误
            if isIRTransformationError(err) {
                // IR转换错误，不触发熔断
                return err
            }
            if isProviderSpecificError(err) {
                // Provider特定错误，触发该provider熔断
                cb.recordFailure(provider, err)
                return err
            }
        }
        cb.recordSuccess(provider)
        return nil
    })
}

func (cb *IRCircuitBreaker) getOrCreateBreaker(provider string) *CircuitBreaker {
    if b, ok := cb.breakers[provider]; ok {
        return b
    }

    // 创建新breaker
    b := NewCircuitBreaker(CircuitBreakerConfig{
        MaxFailures:   5,
        ResetTimeout:  30 * time.Second,
        HalfOpenMax:   3,
    })
    cb.breakers[provider] = b
    return b
}
```
