#### 队列拥堵根因诊断算法

```typescript
// 智能诊断逻辑
function diagnoseQueueBottleneck(snapshot: QueueSnapshot): BottleneckDiagnosis {
  const totalDepth = snapshot.total_queue.depth;
  const modelDepths = snapshot.model_queues.map(q => q.depth);
  const nodeDepths = snapshot.node_queues.map(q => q.depth);
  
  // 1. 如果总队列深度高，但模型队列深度低 → 路由慢
  if (totalDepth > 50 && Math.max(...modelDepths) < 10) {
    return {
      bottleneck: 'routing',
      message: '路由决策慢，考虑优化 route 阶段',
      suggestion: '检查 router.SelectModel() 耗时',
    };
  }
  
  // 2. 如果某个模型队列深度高 → 该模型节点不足
  const topModel = modelDepths.indexOf(Math.max(...modelDepths));
  if (modelDepths[topModel] > 30) {
    return {
      bottleneck: 'model_capacity',
      message: `模型 ${snapshot.model_queues[topModel].model} 节点不足`,
      suggestion: '增加该模型的供应商节点',
    };
  }
  
  // 3. 如果某个节点队列深度高 → 该节点并发不足
  const topNode = nodeDepths.indexOf(Math.max(...nodeDepths));
  if (nodeDepths[topNode] > 20) {
    return {
      bottleneck: 'node_concurrency',
      message: `节点 ${snapshot.node_queues[topNode].credential_id} 并发不足`,
      suggestion: '提高该节点的 max_concurrency 或禁用该节点',
    };
  }
  
  return { bottleneck: 'none', message: '队列正常' };
}
```

  COUNT(DISTINCT ss.session_key) as session_count,
  -- 时间范围
  MIN(ss.first_request_at) as first_activity_at,
  MAX(ss.last_request_at) as last_activity_at,
  -- 成本和请求
  SUM(ss.request_count) as total_requests,
  SUM(ss.total_cost_usd) as total_cost_usd,
  SUM(ss.total_tokens) as total_tokens,
  -- 健康度
  AVG(ss.health_score) as avg_health_score,
  -- 活跃度
  CASE 
    WHEN MAX(ss.last_request_at) > NOW() - INTERVAL '1 hour' THEN 'active'
    WHEN MAX(ss.last_request_at) > NOW() - INTERVAL '1 day' THEN 'recent'
    ELSE 'inactive'
  END as activity_status
FROM session_summaries ss
LEFT JOIN users u ON u.id = ss.user_id
WHERE ss.user_id IS NOT NULL
GROUP BY ss.tenant_id, ss.user_id, u.username, u.email;

```typescript
// 健康度评分算法（0.00 ~ 1.00）
function calculateHealthScore(session: SessionSummary): number {
  // 1. 成功率权重 50%
  const successRate = session.success_count / session.request_count;
  const successScore = successRate * 0.5;
  
  // 2. 延迟权重 30% (延迟越低分数越高)
  const latencyScore = Math.max(0, 1 - session.avg_latency_ms / 5000) * 0.3;
  
  // 3. 成本效率权重 20% (成本越低分数越高，但有底线)
  const costPerToken = session.total_cost_usd / session.total_tokens;
  const costScore = Math.max(0, 1 - costPerToken / 0.00005) * 0.2;
  
  return Math.min(1.0, successScore + latencyScore + costScore);
}
```

**健康度等级**:
- 🟢 **健康** (0.90 ~ 1.00): 成功率高、延迟低、成本合理
- 🟡 **一般** (0.70 ~ 0.89): 偶尔失败或延迟较高
- 🔴 **异常** (< 0.70): 频繁失败或成本过高

    }

        WHERE tenant_id = $1 AND session_key = $2
    `, req.TenantID, req.SessionKey)
    
    // 4. 重新计算健康度

    return nil

func (w *TurnWriter) getCurrentTurnCount(sessionKey string) (int, error) {
    // 先查 Redis 缓存
    key := fmt.Sprintf("session:turn_count:%s", sessionKey)
    count, err := w.redis.Get(context.Background(), key).Int()
    if err == nil {
        return count, nil
    }
    
    // Redis 未命中，查 DB（hot + 冷表）
    var dbCount int
    err = w.db.QueryRow(`
        SELECT COALESCE(MAX(turn_number), 0) 
        FROM v_session_turns_all 
        WHERE session_key = $1
    `, sessionKey).Scan(&dbCount)
    
    if err != nil {
        return 0, err
    }
    
    // 写回 Redis（TTL 1 小时）
    w.redis.Set(context.Background(), key, dbCount, time.Hour)
    
    return dbCount, nil
}
