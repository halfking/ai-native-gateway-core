### Step 1: 收集 24 小时完整数据

```bash
ssh -i "$SSH_KEY_154" -p "$SSH_PORT_154" "$SSH_HOST_154" "
  echo '=== 24小时完整数据统计 ==='
  echo '时间范围: $(date -d \"24 hours ago\" \"+%Y-%m-%d %H:%M\") - $(date \"+%Y-%m-%d %H:%M\")'
  echo ''

  # 1. 新逻辑触发统计
  echo '[1] 新逻辑触发统计（累计）:'
  COOLDOWN_EXT=\$(journalctl -u llm-gateway-go --since '24 hours ago' --no-pager | grep -c 'cooldown_extended_due_to_failure' || echo 0)
  ACTUAL_RECOVER=\$(journalctl -u llm-gateway-go --since '24 hours ago' --no-pager | grep -c 'recovered_with_actual_success' || echo 0)
  AFTER_RECOVER_FAIL=\$(journalctl -u llm-gateway-go --since '24 hours ago' --no-pager | grep -c 'after_recovery' || echo 0)

  echo \"  冷却期延长: \$COOLDOWN_EXT 次\"
  echo \"  实际流量恢复: \$ACTUAL_RECOVER 次\"
  echo \"  恢复后再失败: \$AFTER_RECOVER_FAIL 次\"

  if [ \$COOLDOWN_EXT -gt 0 ]; then
    RATIO=\$(awk \"BEGIN {printf \\\"%.2f\\\", \$ACTUAL_RECOVER / \$COOLDOWN_EXT}\")
    echo \"  恢复/延长比例: \$RATIO (预期约 0.5)\"
  fi
  echo ''

  # 2. 整体请求统计
  echo '[2] 整体请求统计:'
  LOGS=\$(journalctl -u llm-gateway-go --since '24 hours ago' --no-pager)
  TOTAL=\$(echo \"\$LOGS\" | grep -c '\"msg\":\"http_request\"' || echo 0)
  SUCCESS=\$(echo \"\$LOGS\" | grep '\"msg\":\"http_request\"' | grep -c '\"status\":200' || echo 0)
  CLIENT_ERR=\$(echo \"\$LOGS\" | grep '\"msg\":\"http_request\"' | grep -cE '\"status\":4[0-9]{2}' || echo 0)
  SERVER_ERR=\$(echo \"\$LOGS\" | grep '\"msg\":\"http_request\"' | grep -cE '\"status\":5[0-9]{2}' || echo 0)

  echo \"  总请求: \$TOTAL\"
  echo \"  成功 (200): \$SUCCESS\"
  echo \"  客户端错误 (4xx): \$CLIENT_ERR\"
  echo \"  服务端错误 (5xx): \$SERVER_ERR\"

  if [ \$TOTAL -gt 0 ]; then
    SUCCESS_RATE=\$(awk \"BEGIN {printf \\\"%.2f\\\", (\$SUCCESS / \$TOTAL) * 100}\")
    ERROR_RATE=\$(awk \"BEGIN {printf \\\"%.2f\\\", (\$SERVER_ERR / \$TOTAL) * 100}\")
    echo \"  成功率: \${SUCCESS_RATE}%\"
    echo \"  5xx错误率: \${ERROR_RATE}%\"
  fi
  echo ''

  # 3. 节点禁用统计
  echo '[3] 节点禁用统计:'
  DISABLED_FIRST=\$(echo \"\$LOGS\" | grep 'consecutive_3_failures\"' | grep -cv 'after_recovery' || echo 0)
  DISABLED_AFTER=\$(echo \"\$LOGS\" | grep -c 'consecutive_3_failures_after_recovery' || echo 0)

  echo \"  首次禁用: \$DISABLED_FIRST 次\"
  echo \"  恢复后再禁用: \$DISABLED_AFTER 次\"
  echo \"  总禁用: \$((\$DISABLED_FIRST + \$DISABLED_AFTER)) 次\"
"
```

### Step 2: 按小时统计趋势

```bash
ssh -i "$SSH_KEY_154" -p "$SSH_PORT_154" "$SSH_HOST_154" "
  echo '=== 按小时趋势分析 ==='

  for HOUR in {23..0}; do
    START=\"\${HOUR} hours ago\"
    END=\"\$((\$HOUR - 1)) hours ago\"

    if [ \$HOUR -eq 0 ]; then
      END=\"now\"
    fi

    LOGS=\$(journalctl -u llm-gateway-go --since \"\$START\" --until \"\$END\" --no-pager 2>/dev/null || echo '')

    if [ -n \"\$LOGS\" ]; then
      TOTAL=\$(echo \"\$LOGS\" | grep -c '\"msg\":\"http_request\"' || echo 0)
      SUCCESS=\$(echo \"\$LOGS\" | grep '\"msg\":\"http_request\"' | grep -c '\"status\":200' || echo 0)

      if [ \$TOTAL -gt 0 ]; then
        SUCCESS_RATE=\$(awk \"BEGIN {printf \\\"%.1f\\\", (\$SUCCESS / \$TOTAL) * 100}\")
        TIMESTAMP=\$(date -d \"\$START\" '+%H:00')
        echo \"\$TIMESTAMP  请求:\$TOTAL  成功率:\${SUCCESS_RATE}%\"
      fi
    fi
  done
" | tee /tmp/hourly-trends-24h.txt
```
