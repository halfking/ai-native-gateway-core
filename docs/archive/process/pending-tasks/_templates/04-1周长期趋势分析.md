```bash
  ACTUAL_RECOVER=\$(journalctl -u llm-gateway-go --since '7 days ago' --no-pager | grep -c 'recovered_with_actual_success' || echo 0)
  AFTER_RECOVER_FAIL=\$(journalctl -u llm-gateway-go --since '7 days ago' --no-pager | grep -c 'after_recovery' || echo 0)

  echo '[1] 新逻辑触发（7天累计）:'
  echo \"  冷却期延长: \$COOLDOWN_EXT 次 (平均 \$((\$COOLDOWN_EXT / 7)) 次/天)\"
  echo \"  实际流量恢复: \$ACTUAL_RECOVER 次 (平均 \$((\$ACTUAL_RECOVER / 7)) 次/天)\"

  # 整体成功率
  LOGS=\$(journalctl -u llm-gateway-go --since '7 days ago' --no-pager)
  TOTAL=\$(echo \"\$LOGS\" | grep -c '\"msg\":\"http_request\"' || echo 0)
  SUCCESS=\$(echo \"\$LOGS\" | grep '\"msg\":\"http_request\"' | grep -c '\"status\":200' || echo 0)

  echo ''
  echo '[2] 整体请求（7天累计）:'
  echo \"  总请求: \$TOTAL (平均 \$((\$TOTAL / 7)) 次/天)\"
  echo \"  成功: \$SUCCESS\"
  if [ \$TOTAL -gt 0 ]; then
    SUCCESS_RATE=\$(awk \"BEGIN {printf \\\"%.2f\\\", (\$SUCCESS / \$TOTAL) * 100}\")
    echo \"  成功率: \${SUCCESS_RATE}%\"
  fi
"
```

### Step 2: 按天趋势分析

```bash
ssh -i "$SSH_KEY_154" -p "$SSH_PORT_154" "$SSH_HOST_154" "
  echo '=== 按天趋势分析 ==='
  echo '日期,总请求,成功率,冷却期延长,实际恢复'

  for DAY in {6..0}; do
    START=\"\${DAY} days ago\"
    END=\"\$((\$DAY - 1)) days ago\"
    if [ \$DAY -eq 0 ]; then
      END=\"now\"
    fi

    LOGS=\$(journalctl -u llm-gateway-go --since \"\$START\" --until \"\$END\" --no-pager 2>/dev/null)

    TOTAL=\$(echo \"\$LOGS\" | grep -c '\"msg\":\"http_request\"' || echo 0)
    SUCCESS=\$(echo \"\$LOGS\" | grep '\"msg\":\"http_request\"' | grep -c '\"status\":200' || echo 0)
    COOLDOWN=\$(echo \"\$LOGS\" | grep -c 'cooldown_extended' || echo 0)
    RECOVER=\$(echo \"\$LOGS\" | grep -c 'recovered_with_actual' || echo 0)

    if [ \$TOTAL -gt 0 ]; then
      SUCCESS_RATE=\$(awk \"BEGIN {printf \\\"%.1f\\\", (\$SUCCESS / \$TOTAL) * 100}\")
      DATE=\$(date -d \"\$START\" '+%m-%d')
      echo \"\$DATE,\$TOTAL,\${SUCCESS_RATE}%,\$COOLDOWN,\$RECOVER\"
    fi
  done
" | tee /tmp/weekly-trends.csv
```

### Step 3: P1 优化可行性评估

根据数据评估以下 P1 优化方案：

#### P1-A: 探测成功缩短冷却期

**触发条件**: Provider 级别探测成功 → 将该 Provider 所有节点的冷却期从 5 分钟缩短到 30 秒

**数据判断**:
```bash
# 检查商汤节点的恢复模式
ssh -i "$SSH_KEY_154" -p "$SSH_PORT_154" "$SSH_HOST_154" "
  journalctl -u llm-gateway-go --since '7 days ago' --no-pager | \
    grep 'provider_id\":15' | \
    grep -E '(recovered_with_actual|cooldown_extended)' | \
    wc -l
"

# 如果 cooldown_extended 次数多 (>20次/周)，说明需要更快的恢复机制
# 推荐指数: 高
```

#### P1-B: 渐进恢复（三阶段）

**设计**: `disabled` → `probe_recovery`(10%流量) → `verified_recovery`(50%流量) → `normal`

**数据判断**:
```bash
# 检查恢复后再失败的比例
AFTER_FAIL=$(ssh -i "$SSH_KEY_154" -p "$SSH_PORT_154" "$SSH_HOST_154" \
  "journalctl -u llm-gateway-go --since '7 days ago' --no-pager | grep -c 'after_recovery'")

ACTUAL_RECOVER=$(ssh -i "$SSH_KEY_154" -p "$SSH_PORT_154" "$SSH_HOST_154" \
  "journalctl -u llm-gateway-go --since '7 days ago' --no-pager | grep -c 'recovered_with_actual'")

# 如果 AFTER_FAIL / ACTUAL_RECOVER > 0.2 (20%)，说明恢复后仍不稳定
# 推荐指数: 中等
```
