# 智能绑定策略

> 三级绑定体系，平衡厂商缓存利用与成本优化、故障切换。

## 1. 绑定级别定义

### 1.1 SessionRouteBinding

```go
type SessionRouteBinding struct {
    SessionID       string
    StdModel        string
    CredentialID    int
    ProviderID      int
    RawModel        string
    FPSlotIndex     int

    BindingLevel        int       // 1=强, 2=软, 3=无
    BoundAt             time.Time
    LastSuccessAt       time.Time
    LastEvalAt          time.Time
    ConsecutiveSuccess  int

    // 性能基线 (用于软绑定评估)
    BaselineLatencyMs   int
    BaselinePrice       float64
    BaselineSuccessRate float64

    // 请求特征 (决定绑定级别)
    AvgPromptTokens     int
    HasCachedTokens     bool
    TotalRequests       int

    // 版本号 (CAS 防止并发覆盖)
    Version             int
    Invalidated         bool
    InvalidateReason    string
}
```

### 1.2 三级绑定

| 级别 | 触发条件 | 锁定范围 | 切换策略 | 适用模型 |
|------|----------|----------|----------|----------|
| L1 强绑定 | prompt>2000 或 cached>0 或 交互式会话 | credential+provider+rawModel+fpSlot | 仅上游错误 | claude-sonnet, gpt-4o, gemini-1.5 |
| L2 软绑定 | prompt<2000 且无缓存 | credential+provider (rawModel 可漂移) | 上游错误 + 成本劣化 | gpt-4o-mini, 短请求 |
| L3 无绑定 | 首次请求或绑定已失效 | 无 | 每次重新选路 | 首次请求 |

## 2. 绑定级别判定

### 2.1 DetermineBindingLevel

```go
func DetermineBindingLevel(req *Request, resp *Response) int {
    // 优先级: cached > prompt > session attr > default
    if resp.Usage.CachedTokens > 0 && resp.Usage.CachedTokens > resp.Usage.PromptTokens*0.3 {
        return BindingLevelStrong  // 缓存命中, 必须保持
    }
    if req.PromptTokens > 2000 {
        return BindingLevelStrong  // 长上下文
    }
    if req.SessionMetadata.ConversationTurns > 5 {
        return BindingLevelStrong  // 多轮对话
    }
    return BindingLevelSoft        // 默认
}
```

## 4. 软绑定重评估

### 4.1 ShouldReEvaluate

```go
func ShouldReEvaluate(binding *SessionRouteBinding, metrics *NodeMetrics) (bool, string) {
    now := time.Now()

    // 时间窗口: 至少 300s
    if now.Sub(binding.LastEvalAt) < 300*time.Second {
        return false, "too_soon"
    }

    // 请求计数: 每 10 次
    if binding.TotalRequests%10 != 0 {
        return false, "not_due"
    }

    // 新手期: 连续成功 < 5 允许探索
    if binding.ConsecutiveSuccess < 5 {
        return true, "exploration_mode"
    }

    // 错误率
    if metrics.ErrorRate > 0.05 {
        return true, "high_error_rate"
    }

    // 时延劣化
    if metrics.P95LatencyMs > binding.BaselineLatencyMs*1.5 {
        return true, "latency_degraded"
    }

    // 价格劣化
    if metrics.Price > binding.BaselinePrice*1.3 {
        return true, "cost_increased"
    }

    return false, "stable"
}
```

### 4.2 重评估流程

```go
func (m *BindingManager) ReEvaluate(sessionID, stdModel string, candidates []*RouteNode, env *URSMEnvironment) (*SessionRouteBinding, error) {
    // 1. 获取当前 binding
    binding := m.Get(sessionID, stdModel)
    if binding == nil {
        return nil, nil
    }

    // 2. CostScorer 重新评分
    scores := m.costScorer.Score(candidates, env)

    // 3. 找最优候选
    best := findBest(candidates, scores)

    // 4. 对比
    if best.CredentialID == binding.CredentialID {
        // 保持, 更新 LastEvalAt
        binding.LastEvalAt = time.Now()
        m.Update(binding)
        return binding, nil
    }

    // 5. 切换
    costBenefit := estimateSwitchCost(binding, best)
    if costBenefit.ExpectedSavings < costBenefit.SwitchOverhead {
        // 切换成本 > 收益, 保持
        binding.LastEvalAt = time.Now()
        m.Update(binding)
        return binding, nil
    }

    // 6. 执行切换
    oldBinding := binding
    newBinding := m.Create(sessionID, stdModel, best, nil)
    m.Invalidate(sessionID, stdModel, "cost_optimization")
    // Hook: route_changed {from: oldBinding, to: newBinding, reason: cost_optimization}
    return newBinding, nil
}
```

## 5. 绑定切换场景

| 事件 | 强绑定 | 软绑定 |
|------|--------|--------|
| 上游 5xx | 切换 (实际错误) | 切换 |
| 上游 4xx 客户端错误 | 不切换 | 不切换 |
| 上游 429 | 不切换 (冷却) | 不切换 (冷却) |
| 连接超时 | 切换 | 切换 |
| 流中断 (已输出) | 不切换 (已输出) | 不切换 (已输出) |
| 流中断 (未输出) | 切换 | 切换 |
| 凭据 auth 失败 | 切换 + 永久失效 | 切换 + 永久失效 |
| 价格变高 | 不切换 | 重评估时切换 |
| 更优候选出现 | 不切换 | 重评估时切换 |
| 管理员禁用 | 失效 | 失效 |

## 6. Redis 存储

```go
// Key:  ursm:session_route:{sessionID}:{stdModel}
// Value: HSET credential_id, provider_id, raw_model, fp_slot_index, level,
//        version, bound_at, invalidated, invalidate_reason
// TTL:  24h (活跃请求自动刷新)

// Lua 原子创建:
local key = KEYS[1]
local exists = redis.call('EXISTS', key)
if exists == 1 then return 0 end  -- 已有绑定, 不覆盖
redis.call('HMSET', key, unpack(ARGV))
redis.call('EXPIRE', key, 86400)
return 1

// Lua CAS 更新:
local key = KEYS[1]
local version = redis.call('HGET', key, 'version')
if version ~= ARGV[1] then return 0 end  -- 版本不匹配
redis.call('HMSET', key, unpack(ARGV, 2))
return 1
```
