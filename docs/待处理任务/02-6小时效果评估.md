### Step 1: 统计 6 小时数据

```bash
ssh -i "$SSH_KEY_154" -p "$SSH_PORT_154" "$SSH_HOST_154" "
  echo '=== 6小时运行数据统计 ==='
  echo '时间范围: $(date -d \"6 hours ago\" \"+%Y-%m-%d %H:%M\") - $(date \"+%Y-%m-%d %H:%M\")'
  echo ''

  echo '[1] 新逻辑触发次数:'
  COOLDOWN_EXT=\$(journalctl -u llm-gateway-go --since '6 hours ago' --no-pager | \
    grep -c 'cooldown_extended_due_to_failure' || echo 0)
  ACTUAL_RECOVER=\$(journalctl -u llm-gateway-go --since '6 hours ago' --no-pager | \
    grep -c 'recovered_with_actual_success' || echo 0)
  AFTER_RECOVER_FAIL=\$(journalctl -u llm-gateway-go --since '6 hours ago' --no-pager | \
    grep -c 'after_recovery' || echo 0)

  echo \"  冷却期延长: \$COOLDOWN_EXT 次\"
  echo \"  实际流量恢复: \$ACTUAL_RECOVER 次\"
  echo \"  恢复后再失败: \$AFTER_RECOVER_FAIL 次\"
  echo ''

  echo '[2] 请求统计:'
  LOGS=\$(journalctl -u llm-gateway-go --since '6 hours ago' --no-pager)
  TOTAL=\$(echo \"\$LOGS\" | grep -c '\"msg\":\"http_request\"' || echo 0)
  SUCCESS=\$(echo \"\$LOGS\" | grep '\"msg\":\"http_request\"' | grep -c '\"status\":200' || echo 0)
  ERRORS=\$(echo \"\$LOGS\" | grep '\"msg\":\"http_request\"' | grep -cE '\"status\":(4|5)[0-9]{2}' || echo 0)

  echo \"  总请求: \$TOTAL\"
  echo \"  成功: \$SUCCESS\"
  echo \"  错误: \$ERRORS\"
  if [ \$TOTAL -gt 0 ]; then
    SUCCESS_RATE=\$(awk \"BEGIN {printf \\\"%.2f\\\", (\$SUCCESS / \$TOTAL) * 100}\")
    echo \"  成功率: \${SUCCESS_RATE}%\"
  fi
  echo ''

  echo '[3] 节点禁用统计:'
  DISABLED_COUNT=\$(echo \"\$LOGS\" | grep -c '\"disabled\":true' || echo 0)
  echo \"  节点禁用次数: \$DISABLED_COUNT\"
"
```
