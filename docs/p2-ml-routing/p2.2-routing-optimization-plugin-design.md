### 2.2 插件接口设计

```go
// routingopt/plugin.go

// RoutingOptimizer 是插件的顶层接口
type RoutingOptimizer interface {
	// PreClassify 在分类前增强特征
	PreClassify(ctx context.Context, signals *autoroute.ClassificationSignals) (*EnhancedSignals, error)
	
	// PostClassify 在分类后调整置信度
	PostClassify(ctx context.Context, taskType autoroute.TaskType, confidence float64) (adjustedConfidence float64, err error)
	
	// RecommendModel 重排序候选模型
	RecommendModel(ctx context.Context, candidates []ModelCandidate, context RoutingContext) ([]ModelCandidate, error)
	
	// RecordFeedback 记录路由结果反馈
	RecordFeedback(ctx context.Context, feedback RoutingFeedback) error
	
	// GetStats 获取优化统计
	GetStats(ctx context.Context) (*OptimizerStats, error)
}

// EnhancedSignals 增强后的分类信号
type EnhancedSignals struct {
	Original       *autoroute.ClassificationSignals
	UserAffinities map[string]float64  // 用户对各task type的历史亲和力
	SessionPattern *SessionPattern     // 会话模式 (IDE/CLI/Web)
	TimeContext    *TimeContext        // 时间上下文 (峰值/非峰值)
}

// ModelCandidate 候选模型
type ModelCandidate struct {
	Provider      string
	Model         string
	Score         float64
	LatencyP99    time.Duration
	CostPerToken  float64
	QualityScore  float64
	AvailableRate float64
}

// RoutingContext 路由决策上下文
type RoutingContext struct {
	TaskType       autoroute.TaskType
	Profile        string
	UserID         *string
	SessionID      *string
	PreviousRoutes []string  // 最近3次路由历史
	Constraints    *Constraints
}

// Constraints 路由约束
type Constraints struct {
	MaxLatency    *time.Duration
	MaxCost       *float64
	MinQuality    *float64
	PreferredTier *string  // "free", "low", "mid", "high"
}

// RoutingFeedback 路由反馈
type RoutingFeedback struct {
	RequestID      string
	TaskType       autoroute.TaskType
	PredictedProvider string
	ActualLatency  time.Duration
	ActualCost     float64
	Success        bool
	ErrorType      *string
	UserCorrection *UserCorrection  // 来自人工标注
}

// UserCorrection 人工纠正
type UserCorrection struct {
	CorrectProvider string
	Reason          string  // performance/cost/availability/quality
	Annotator       string
	AnnotatedAt     time.Time
}
```

## 五、关键技术细节

### 5.1 插件注入方式

```go
// autoroute/index.go (现有文件, 修改)

type Decider struct {
	classifier Classifier
	index      *Index
	// ... 现有字段
	
	// 新增: 插件字段 (nil = 插件禁用)
	optimizer routingopt.RoutingOptimizer
}

func (d *Decider) Decide(ctx context.Context, signals ClassificationSignals, ...) (*Decision, error) {
	// 1. 插件预处理 (增强特征)
	var enhancedSignals *routingopt.EnhancedSignals
	if d.optimizer != nil {
		enhanced, err := d.optimizer.PreClassify(ctx, &signals)
		if err != nil {
			// 降级: 使用原始signals
			log.Warn("optimizer.PreClassify failed, using original signals", "error", err)
		} else {
			enhancedSignals = enhanced
		}
	}
	
	// 2. 分类 (使用增强后的signals)
	taskType, confidence, err := d.classifier.Classify(ctx, signals)
	if err != nil {
		return nil, err
	}
	
	// 3. 插件后处理 (调整置信度)
	if d.optimizer != nil {
		adjustedConf, err := d.optimizer.PostClassify(ctx, taskType, confidence)
		if err == nil {
			confidence = adjustedConf
		}
	}
	
	// 4. 获取候选模型 (现有逻辑)
	candidates := d.getCandidates(ctx, taskType, profile)
	
	// 5. 插件重排序
	if d.optimizer != nil {
		routingCtx := routingopt.RoutingContext{
			TaskType: taskType,
			Profile: profile,
			// ...
		}
		reranked, err := d.optimizer.RecommendModel(ctx, candidates, routingCtx)
		if err == nil {
			candidates = reranked
		}
	}
	
	// 6. 选择最佳候选
	chosen := candidates[0]
	
	// 7. 记录反馈 (异步)
	if d.optimizer != nil {
		go func() {
			feedback := routingopt.RoutingFeedback{
				RequestID: requestID,
				TaskType: taskType,
				PredictedProvider: chosen.Provider,
				// ...
			}
			_ = d.optimizer.RecordFeedback(context.Background(), feedback)
		}()
	}
	
	return &Decision{Provider: chosen.Provider, ...}, nil
}
```

### 5.2 特征增强算法

```go
// routingopt/classifier_enhancer.go

func (e *ClassificationEnhancer) Enhance(ctx context.Context, signals *autoroute.ClassificationSignals) (*EnhancedSignals, error) {
	enhanced := &EnhancedSignals{
		Original: signals,
	}
	
	// 1. 用户亲和力 (Redis缓存, 100ms TTL)
	if userID := extractUserID(ctx); userID != "" {
		affinity, err := e.userAffinityCache.Get(ctx, userID)
		if err == nil {
			enhanced.UserAffinities = affinity.TaskTypeDistribution
		}
	}
	
	// 2. 会话模式识别
	sessionID := extractSessionID(ctx)
	pattern := e.detectSessionPattern(ctx, sessionID, signals)
	enhanced.SessionPattern = pattern
	
	// 3. 时间上下文
	now := time.Now()
	isPeakHour := (now.Hour() >= 9 && now.Hour() <= 18)  // 工作时间
	enhanced.TimeContext = &TimeContext{
		IsPeakHour: isPeakHour,
		Timestamp: now,
	}
	
	return enhanced, nil
}

func (e *ClassificationEnhancer) detectSessionPattern(ctx context.Context, sessionID string, signals *autoroute.ClassificationSignals) *SessionPattern {
	// 查询最近10次该session的请求
	recentReqs, _ := e.sessionPatternDB.GetRecent(ctx, sessionID, 10)
	
	codeCount := 0
	reasoningCount := 0
	chatCount := 0
	
	for _, req := range recentReqs {
		switch req.TaskType {
		case "code":
			codeCount++
		case "reasoning":
			reasoningCount++
		case "chat":
			chatCount++
		}
	}
	
	// 模式分类
	if codeCount >= 7 {
		return &SessionPattern{Type: "ide", Confidence: 0.9}
	} else if reasoningCount+codeCount >= 7 {
		return &SessionPattern{Type: "cli", Confidence: 0.8}
	} else if chatCount >= 5 {
		return &SessionPattern{Type: "web", Confidence: 0.7}
	}
```

```go
// routingopt/model_recommender.go

func (r *ModelRecommender) Score(ctx context.Context, candidate ModelCandidate, context RoutingContext) float64 {
	// 基础指标 (归一化到0-1)
	qualityNorm := candidate.QualityScore  // 已归一化
	costNorm := 1.0 - math.Min(candidate.CostPerToken/0.01, 1.0)  // 0.01为最贵
	latencyNorm := 1.0 - math.Min(float64(candidate.LatencyP99)/float64(5*time.Second), 1.0)  // 5s为最慢
	availabilityNorm := candidate.AvailableRate  // 已归一化
	
	// 动态权重 (基于profile和时间)
	weights := r.getDynamicWeights(context)
	
	// 加权评分
	score := weights.Quality*qualityNorm -
		weights.Cost*costNorm +
		weights.Latency*latencyNorm +
		weights.Availability*availabilityNorm
	
	// 人工标注增强: 如果该provider有高准确率, 提升评分
	if humanAccuracy, ok := r.getHumanAnnotationAccuracy(context.TaskType, candidate.Provider); ok {
		if humanAccuracy > 0.8 {
			score *= 1.1  // 10%提升
		} else if humanAccuracy < 0.5 {
			score *= 0.9  // 10%惩罚
		}
	}
	
	return score
}

func (r *ModelRecommender) getDynamicWeights(context RoutingContext) Weights {
	// 默认权重
	w := Weights{
		Quality: 0.4,
		Cost: 0.3,
		Latency: 0.2,
		Availability: 0.1,
	}
	
	// Profile调整
	switch context.Profile {
	case "free":
		w.Cost = 0.5  // 成本优先
		w.Quality = 0.2
	case "premium":
		w.Quality = 0.6  // 质量优先
		w.Cost = 0.1
	case "balanced":
		// 保持默认
	}
	
	// 时间调整 (高峰期优先free tier)
	if context.TimeContext != nil && context.TimeContext.IsPeakHour {
		w.Cost += 0.1
		w.Quality -= 0.1
	}
	
	// 任务类型调整
	switch context.TaskType {
	case autoroute.TaskReasoning:
		w.Quality = 0.6  // reasoning需要高质量
		w.Latency = 0.1
	case autoroute.TaskCode:
		w.Quality = 0.5
		w.Latency = 0.2  // code需要较快响应
	case autoroute.TaskChat:
		w.Latency = 0.4  // chat优先低延迟
		w.Quality = 0.3
	}
	
	return w
}
```

```go
// routingopt/adaptive_learner.go

func (l *AdaptiveLearner) Update(ctx context.Context) error {
	// 1. 获取最近1000次反馈
	feedbacks, err := l.feedbackRepo.GetRecent(ctx, 1000)
	if err != nil {
		return err
	}
	
	// 2. 计算当前准确率
	currentAccuracy := l.calculateAccuracy(feedbacks)
	
	// 3. 加载上一版本参数
	prevState, err := l.paramStore.GetActive(ctx)
	if err != nil {
		return err
	}
	
	// 4. 检测异常 (准确率下降>5%)
	if currentAccuracy < prevState.OverallAccuracy-0.05 {
		log.Warn("accuracy drop detected", "prev", prevState.OverallAccuracy, "current", currentAccuracy)
		// 触发告警
		l.sendAlert(ctx, "accuracy_drop", currentAccuracy)
	}
	
	// 5. 决定是否更新参数
	if currentAccuracy > prevState.OverallAccuracy+0.02 {  // 提升>2%才更新
		// 使用Bayesian Optimization调整参数
		newParams := l.optimizeParameters(ctx, feedbacks)
		
		// 保存新版本
		newState := &RoutingOptimizationState{
			Version: prevState.Version + 1,
			ClassifierWeights: newParams.ClassifierWeights,
			RecommenderWeights: newParams.RecommenderWeights,
			OverallAccuracy: currentAccuracy,
			ActivatedAt: time.Now(),
		}
		
		err = l.paramStore.Save(ctx, newState)
		if err != nil {
			return err
		}
		
		// 停用旧版本
		prevState.DeactivatedAt = time.Now()
		_ = l.paramStore.Update(ctx, prevState)
		
		log.Info("parameters updated", "version", newState.Version, "accuracy", currentAccuracy)
	}
	
	return nil
}

func (l *AdaptiveLearner) calculateAccuracy(feedbacks []RoutingFeedback) float64 {
	correct := 0
	total := 0
	
	for _, fb := range feedbacks {
		if fb.HasHumanCorrection {
			// 人工标注: 权重x2
			total += 2
			if fb.PredictedProvider == fb.CorrectProvider {
				correct += 2
			}
		} else if fb.Success {
			// 自动反馈: 成功视为正确
			total++
			correct++
		} else {
			// 自动反馈: 失败视为可能错误
			total++
		}
	}
	
	if total == 0 {
		return 0
	}
	
	return float64(correct) / float64(total)
}
```

```go
// 自动回滚触发条件
func (l *AdaptiveLearner) shouldRollback(currentState, prevState *RoutingOptimizationState) bool {
	// 1. 准确率下降>10%
	if currentState.OverallAccuracy < prevState.OverallAccuracy-0.1 {
		return true
	}
	
	// 2. 错误率飙升
	if currentState.ErrorRate > 0.05 {  // >5%错误率
		return true
	}
	
	// 3. P99延迟增加>100ms
	if currentState.P99LatencyMs > prevState.P99LatencyMs+100 {
		return true
	}
	
	return false
}

// 执行回滚
func (l *AdaptiveLearner) Rollback(ctx context.Context, targetVersion int) error {
	// 1. 加载目标版本参数
	targetState, err := l.paramStore.GetByVersion(ctx, targetVersion)
	if err != nil {
		return err
	}
	
	// 2. 停用当前版本
	currentState, _ := l.paramStore.GetActive(ctx)
	currentState.DeactivatedAt = time.Now()
	_ = l.paramStore.Update(ctx, currentState)
	
	// 3. 激活目标版本
	targetState.ActivatedAt = time.Now()
	targetState.DeactivatedAt = nil
	err = l.paramStore.Update(ctx, targetState)
	if err != nil {
		return err
	}
	
	// 4. 热重载参数 (无需重启服务)
	l.reloadParameters(targetState)
	
	log.Warn("rolled back to version", "version", targetVersion)
	return nil
}
```
